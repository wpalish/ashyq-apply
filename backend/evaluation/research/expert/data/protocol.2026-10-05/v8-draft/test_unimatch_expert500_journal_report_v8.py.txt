def prepare_reporting(
    pilot,
    report,
    *,
    failed=False,
    rejected=False,
    minimal=False,
    next_source=False,
    stored_view=False,
    declined=False,
    failed_kind="selector_timeout",
    next_action=None,

    bootstrap(pilot)

                "tool": "view",

            lambda prompt: decision(

            "synthetic_fixture": True,

    )
    write(baseline_dir / "run.json", {"run_id": 777, "headSha": HEAD, "conclusion": "success"})
    scripts = {
        str(path): report["file_hash"](path)
        for path in (
            PILOT,
            REPORT,
            Path("/tmp/unimatch-expert500-pilot-v1-frozen.py"),

    return report["verify_journal_replay"](
        pilot.folder,
        pilot.read("trace.json"),
        pilot.packet,
        pilot.controller_dirs[-1] / "artifact/logs",
        pilot.runtime["validate_all_fact_citations"],
    )


def run_without_labels(pilot, report, monkeypatch):
    """Exercise the real report main while its scorer sees no labels at all."""
    dataset = pilot.root / "no-labels-sentinel.txt"
    dataset.write_text("SYNTHETIC_NO_REVIEWED_LABELS")
    reads = []

    class NoLabels:
        @classmethod
        def model_validate_json(cls, value):
            if isinstance(value, bytes):
                value = value.decode()
            assert value == "SYNTHETIC_NO_REVIEWED_LABELS"
            assert (pilot.root / "submission-seal.json").is_file()
            reads.append(value)
            return None

    def no_label_score(dataset, capture):
        assert dataset is None
        return {
            "metrics": {
                "claim_recall": {"numerator": 0, "denominator": 0, "value": None},
                "claim_precision": {"numerator": 0, "denominator": 0, "value": None},
            },
            "cases": [{"case_id": observation.case_id} for observation in capture.observations],
        }

    report.update(Dataset=NoLabels, score=no_label_score)
    monkeypatch.setattr(
        sys,
        "argv",
        [
            "journal-report",
            "--root",
            str(pilot.root),
            "--dataset",
            str(dataset),
            "--orchestrator",
            str(PILOT),
        ],
    )
    return reads


def persist_result(pilot, result):
    write(pilot.folder / "result.json", result)
    write(pilot.root / "experts-results.json", {"stopped": False, "results": [result]})

    result = pilot.read("result.json")
    result["navigation_outcomes"] = outcomes
    persist_result(pilot, result)

def assert_prelabel_rejection(pilot, report, monkeypatch):
    reads = run_without_labels(pilot, report, monkeypatch)
    with pytest.raises((ValueError, FileNotFoundError)):
        report["main"]()
    assert reads == [] and not (pilot.root / "submission-seal.json").exists()

def test_three_digit_turns_replay_original_sources_in_numeric_order(pilot, report, monkeypatch):
    from test_unimatch_expert500_journal_pilot_v8 import source_action

    # Rejected complete choices, then saved views, are permitted turns and use no
    # additional network budget. The original source runs are genuinely20/100.
    for _ in range(19):
        pilot.queue(lambda prompt: decision(prompt, "submit"), elapsed=1)
    pilot.queue(elapsed=1)
    for _ in range(79):
        pilot.queue(lambda prompt: decision(prompt, action=source_action("view")), elapsed=1)
    result = prepare_reporting(pilot, report)
    assert [path.name for path in pilot.controller_dirs] == ["controller-20", "controller-100"]
    assert pilot.read("journal-events.json")[-1]["turn"] == 102
    verified = replay(pilot, report)
    assert (
        verified["explicit_commit"] and verified["final_journal_sha256"] == result["journal_sha256"]
    )
    ledger = report["verify_source_ledger"](pilot.folder, pilot.packet, result)
    assert ledger["controller_count"] == 2 and ledger["counters"]["fetcher_calls"] == 2
    reads = run_without_labels(pilot, report, monkeypatch)
    report["main"]()
    assert reads == ["SYNTHETIC_NO_REVIEWED_LABELS"]
    sealed = json.loads((pilot.root / "submission-seal.json").read_text())
    assert sealed

@pytest.mark.parametrize(
    "change",
    ["tool", "failed", "unfinished", "second_completion", "different_answer", "duplicate_json"],
)
def test_original_selector_provenance_is_required_for_checkpoint_replay(pilot, report, change):
    prepare_reporting(pilot, report)
    path = pilot.folder / "selector-02.events.jsonl"
    raw = (pilot.folder / "selector-02.answer.json").read_text()
    rows = [json.loads(line) for line in completed_events(raw).splitlines()]
    if change == "tool":
        rows.insert(
            2,
            {
                "type": "item.completed",
                "item": {"id": "tool", "type": "command_execution", "command": "cat sealed"},
            },
        )
    elif change == "failed":
        rows[-1] = {"type": "turn.failed", "error": {"message": "Synthetic failure"}}
    elif change == "unfinished":
        rows.pop()
    elif change == "second_completion":
        rows.append(copy.deepcopy(rows[-1]))
    elif change == "different_answer":
        changed = json.loads(raw)
        changed["observation"] = "A different completed decision."
        rows[2]["item"]["text"] = json.dumps(changed)
    else:
        duplicate = raw.replace('"kind": "record"', '"kind": "record", "kind": "submit"', 1)
        assert duplicate != raw
        (pilot.folder / "selector-02.answer.json").write_text(duplicate)
        rows[2]["item"]["text"] = duplicate
    path.write_text("\n".join(json.dumps(row) for row in rows) + "\n")
    with pytest.raises(ValueError):
        replay(pilot, report)

        result["meter"][0]["active_ms"] += 1
    write(pilot.folder / "journal-events.json", events)
    write(pilot.folder / "result.json", result)
    with pytest.raises(ValueError):

def update_baseline(pilot, *, budget_changes=None, config_changes=None, observation_error=None):
    folder = pilot.root / "baseline/u"
    result_path, capture_path = folder / "result.json", folder / "capture.json"
    baseline = json.loads(result_path.read_text())
    capture = json.loads(capture_path.read_text())
    for key, value in (budget_changes or {}).items():
        if value is None:
            baseline["budget"].pop(key, None)
        else:
            baseline["budget"][key] = value
    for key, value in (config_changes or {}).items():
        if value is None:
            capture["config"].pop(key, None)
        else:
            capture["config"][key] = value
    capture["config"]["case_budgets"]["u"] = copy.deepcopy(baseline["budget"])
    if observation_error is not None:
        capture["observations"][0]["error"] = observation_error
    write(result_path, baseline)
    write(capture_path, capture)
    return baseline["budget"], Capture.model_validate(capture)

def pure_source(*, prior=SourceCounters(), tool="fetcher"):
    started = datetime(2026, 10, 4, tzinfo=UTC)
    guard = WallGuard(started, started + timedelta(milliseconds=WALL_GUARD_MS))
    request = (
        {"tool": tool, "url": URL}
        if tool == "fetcher"
        else {"tool": tool, "query": "official", "domains": ["u.edu"]}
    )
    plan = {
        "case_id": "u",
        "log_prefix": "u",
        "strict_budget": True,
        "max_calls": 60,
        "max_ms": WALL_GUARD_MS,
        "max_search_queries": 6,
        "budget_origin": NETWORK_COMPARISON_MODE,
        "budget_model": BUDGET_MODEL,
        **guard.as_dict(),
        "official_domains": ["u.edu"],
        "actions": [request],
        "start_calls": prior.fetcher_calls,
        "start_search_queries": prior.search_queries,
        "start_ms": prior.source_ms,
    }
    policy = {
        "strict_budget": True,
        "budget_origin": NETWORK_COMPARISON_MODE,
        "exploration_mode": "same_policy",
        "official_domains": ["u.edu"],
        "allowed_search_providers": ["exa", "exa_mcp"],
        **expert_budget(),
        **guard.as_dict(),
    }
    identity = {
        "case_id": "u",
        "provider": "exa_mcp",
        "log_prefix": "u",
        "budget_origin": NETWORK_COMPARISON_MODE,
        **expert_budget(),
        **guard.as_dict(),
    }
    raw = {
        **identity,
        "tool_policy": policy,
        "phase": "actions",
        "request": request,
        "executed": True,
        "at": (started + timedelta(seconds=200)).isoformat(),
        "dispatch_at": (started + timedelta(seconds=200)).isoformat(),
        "completed_at": (started + timedelta(seconds=200, milliseconds=10)).isoformat(),
        "budgets": {"max_calls": 60, "max_ms": WALL_GUARD_MS, "max_search_queries": 6},
        "elapsed_ms": 10,
        "elapsed_ms_before": prior.source_ms,
        "charged_elapsed_ms": 10,
        "fetcher_calls_cumulative": prior.fetcher_calls + int(tool == "fetcher"),
        "search_queries_cumulative": prior.search_queries + int(tool == "search_provider"),
        "elapsed_ms_cumulative": prior.source_ms + 10,
        "outcome_status": "success",
        "response": {
            "outcome": "ok",
            "outcome_status": "success",
            **({"ok": True, "status_code": 200} if tool == "fetcher" else {"results": []}),
        },
        "stop_reason": None,
        "policy_mode": "same_policy",
    }
    action = {
        **{
            key: raw[key]
            for key in (
                "request",
                "at",
                "elapsed_ms",
                "fetcher_calls_cumulative",
                "search_queries_cumulative",
                "elapsed_ms_cumulative",
                "outcome_status",
                "policy_mode",
            )
        },
        "tool": tool,
        "outcome": "ok",
        "executed": True,
        "final_url": None,
        "raw_log_path": "u-001.json",
        "raw_log_sha256": "a" * 64,
    }
    summary = {
        **identity,
        "tool_policy": policy,
        "initial": {key: plan[key] for key in ("start_calls", "start_ms", "start_search_queries")},
        "actions": [action],
        "raw_logs": {"u-001.json": "a" * 64},
        "final": {
            "fetcher_calls": raw["fetcher_calls_cumulative"],
            "search_queries": raw["search_queries_cumulative"],
            "elapsed_ms": raw["elapsed_ms_cumulative"],
            "stop_reason": None,
            "completed_at": raw["completed_at"],
        },
    }
    return plan, summary, raw, guard

@pytest.mark.parametrize(
    "layer",
    [
        "baseline_result",
        "baseline_capture",
        "baseline_dispatch",
        "baseline_budget",
        "expert_result",
        "dispatch_input",
        "dispatch_plan",
        "summary",
        "raw",
        "all_controller_fields",
    ],
)
def test_v8_original_arm_and_controller_receipts_cannot_mix_providers(
    pilot, report, monkeypatch, layer
):
    result = prepare_reporting(pilot, report)
    directory = pilot.controller_dirs[0]
    if layer == "baseline_result":
        path = pilot.root / "baseline/u/result.json"
        value = json.loads(path.read_text())
        value["provider"] = "exa_mcp"
        write(path, value)
    elif layer == "baseline_capture":
        path = pilot.root / "baseline/u/capture.json"
        value = json.loads(path.read_text())
        value["config"]["search_provider"] = "exa_mcp"
        write(path, value)
    elif layer == "baseline_dispatch":
        path = pilot.root / "baseline/u/dispatch.json"
        value = json.loads(path.read_text())
        value["inputs"]["search_provider"] = "exa_mcp"
        write(path, value)
    elif layer == "baseline_budget":
        update_baseline(pilot, budget_changes={"provider": "exa_mcp"})
    elif layer == "expert_result":
        result["provider"] = "exa_mcp"
        persist_result(pilot, result)
    else:
        dispatch_path = directory / "dispatch.json"
        summary_path = directory / "artifact/summary.json"
        dispatch = json.loads(dispatch_path.read_text())
        plan = json.loads(dispatch["inputs"]["plan"])
        summary = json.loads(summary_path.read_text())
        if layer in {"dispatch_input", "all_controller_fields"}:
            dispatch["inputs"]["search_provider"] = "exa_mcp"
        if layer in {"dispatch_plan", "all_controller_fields"}:
            plan["provider"] = "exa_mcp"
        if layer in {"summary", "all_controller_fields"}:
            summary["provider"] = "exa_mcp"
        if layer in {"raw", "all_controller_fields"}:
            path = directory / "artifact/logs" / summary["actions"][0]["raw_log_path"]
            raw = json.loads(path.read_text())
            raw["provider"] = "exa_mcp"
            write(path, raw)
            sha = report["file_hash"](path)
            summary["actions"][0]["raw_log_sha256"] = sha
            summary["raw_logs"][path.name] = sha
        dispatch["inputs"]["plan"] = json.dumps(plan)
        write(dispatch_path, dispatch)
        write(summary_path, summary)
        with pytest.raises(ValueError, match="provider|raw|summary"):
            report["verify_source_ledger"](pilot.folder, pilot.packet, result)
    assert_prelabel_rejection(pilot, report, monkeypatch)

@pytest.mark.parametrize("status", [401, 402, 403, 429])
def test_v8_actual_unaccepted_provider_refusal_tail_is_unscored_even_if_stop_is_erased(
    pilot, report, monkeypatch, status
):
    import test_unimatch_expert500_journal_pilot_v8 as pilot_fixture

    execute = pilot_fixture.execute

    class Refusal(RuntimeError):
        http_status = status

    async def refused(**kwargs):
        raise Refusal("Synthetic provider refusal")

    async def execute_refusal(*args, **kwargs):
        kwargs["search"] = refused
        return await execute(*args, **kwargs)

    monkeypatch.setattr(pilot_fixture, "execute", execute_refusal)
    result = prepare_reporting(
        pilot, report, next_source=True, next_action=pilot_fixture.source_action("search_provider")
    )
    assert (
        result["validated"]
        and result["accepted_source_action_count"] == 1
        and result["actions"] == 2
    )
    assert result["error"] == f"provider_http_{status}" and pilot.runtime["STOP"].is_set()
    assert len(pilot.calls) == 1 and len(pilot.queries) == 0
    raw_path = pilot.controller_dirs[-1] / "artifact/logs"
    summary = json.loads((pilot.controller_dirs[-1] / "artifact/summary.json").read_text())
    raw = json.loads((raw_path / summary["actions"][0]["raw_log_path"]).read_text())
    assert raw["response"]["http_status"] == status
    assert raw["stop_reason"] == ("http_429" if status == 429 else None)
    # prepare_reporting deliberately writes aggregate stopped=False to verify
    # the original raw refusal independently prevents any seal or label load.
    with pytest.raises(ValueError, match=f"HTTP{status}"):
        report["verify_source_ledger"](pilot.folder, pilot.packet, result)
    assert_prelabel_rejection(pilot, report, monkeypatch)
