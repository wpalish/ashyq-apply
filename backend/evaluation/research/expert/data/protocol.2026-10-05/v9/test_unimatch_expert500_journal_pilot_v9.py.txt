def source_action(tool="fetcher", **changes):
    value = {
        "tool": tool,
        "url": URL if tool == "fetcher" else None,
        "query": "site:u.edu official admissions" if tool == "search_provider" else None,
        "domains": ["u.edu"] if tool == "search_provider" else [],
        "max_results": 4 if tool == "search_provider" else None,
        "source_step": 1 if tool == "view" else None,
        "page": None,
        "offset": None,
        "section_offset": None,
        "link_offset": None,
    }
    return {**value, **changes}


def fact(**changes):
    value = {
        "id": "english",
        "question": "Q1",
        "claim_key": "ielts.overall",
        "value_json": "6.5",
        "source_step": 1,
        "quotation": QUOTE,
        "location": {"kind": "text", "block_id": None, "section_path": [], "page": None},
        "scope": dict(SCOPE),
    }
    return {**value, **changes}


def finding(**changes):
    value = {
        "question": "Q1",
        "status": "found",
        "fact_ids": ["english"],
        "scope": dict(SCOPE),
        "observation": "The recorded synthetic page states this minimum.",
        "supporting_steps": [],
    }
    return {**value, **changes}


def decision(prompt, kind="action", facts=None, findings=None, action=None, **changes):
    if "accepted_journal" not in prompt:
        assert kind == "submit" and action is None and not facts and not findings
        return minimal_commit(prompt, **changes)
    value = {
        "kind": kind,
        "observation": "Inspect the current recorded official source.",
        "journal_sha256": prompt["accepted_journal"]["sha256"],
        "action": source_action() if kind == "action" and action is None else action,
        "facts": facts or [],
        "question_findings": findings or [],
    }
    return {**value, **changes}


def minimal_commit(prompt, **changes):
    from evaluation.research.expert.representation import CLOSING_OBSERVATION

    current_hash = (
        prompt["accepted_journal"]["sha256"]
        if "accepted_journal" in prompt
        else prompt["journal_sha256"]
    )
    return {
        "kind": "submit",
        "observation": CLOSING_OBSERVATION,
        "journal_sha256": current_hash,
        "action": None,
        "facts": [],
        "question_findings": [],
        **changes,
    }


def completed_events(answer):
    return (
        "\n".join(
            json.dumps(item)
            for item in [
                {"type": "thread.started", "thread_id": "synthetic"},
                {"type": "turn.started"},
                {
                    "type": "item.completed",
                    "item": {"id": "answer", "type": "agent_message", "text": answer},
                },
                {"type": "turn.completed", "usage": {"input_tokens": 1, "output_tokens": 1}},
            ]
        )
        + "\n"
    )


@dataclass
class Harness:
    runtime: dict
    repo: Path
    root: Path
    workspace: Path
    packet: dict
    pending: list = field(default_factory=list)
    calls: list = field(default_factory=list)
    queries: list = field(default_factory=list)
    prompts: list = field(default_factory=list)
    commands: list = field(default_factory=list)
    plans: list = field(default_factory=list)
    controller_dirs: list = field(default_factory=list)
    summary_hook: object = None
    workflow_error: Exception | None = None
    source_body: bytes | None = None
    source_content_type: str = "text/html"
    plan_hook: object = None
    search_error_status: int = 429

    @property
    def folder(self):
        return self.root / "experts/u"

    def queue(self, callback=None, *, elapsed=12000, preparation=0, error=None, raw=None):
        self.pending.append(
            {
                "callback": callback,
                "elapsed": elapsed,
                "preparation": preparation,
                "error": error,
                "raw": raw,
            }
        )

    def read(self, name):
        return json.loads((self.folder / name).read_text())

    def run(self):
        return self.runtime["expert"](
            self.repo, "synthetic-frozen", self.root, "u", {"u": self.workspace}
        )

def bootstrap(pilot):
    pilot.queue()


def checkpoint(pilot, *, kind="record", elapsed=12000, **changes):
    pilot.queue(
        lambda prompt: decision(prompt, kind, facts=[fact()], findings=[finding()], **changes),
        elapsed=elapsed,
    )


def commit(pilot, *, elapsed=500, acknowledge_source=None):
    pilot.queue(
        lambda prompt: decision(
            prompt,
            "submit",
            facts=[
                fact(id="english_second_source", question="OTHER", source_step=acknowledge_source)
            ]
            if acknowledge_source is not None
            else [],
        ),
        elapsed=elapsed,
    )


def assert_exact_clock(result):
    cumulative = 0
    for entry in result["meter"]:
        assert type(entry["elapsed_ms"]) is int and entry["elapsed_ms"] >= 0
        cumulative += entry["elapsed_ms"]
        assert entry["active_ms"] == cumulative
    assert result["active_ms"] == cumulative


class FakeDatetime(datetime):
    current = datetime.now(UTC)

    @classmethod
    def now(cls, tz=None):
        return cls.current.astimezone(tz or UTC)

@pytest.mark.parametrize("bad", ["quote", "location", "stale_hash", "duplicate_fact"])
def test_network_mode_does_not_weaken_atomic_source_citation_guards(pilot, bad):
    bootstrap(pilot)
    checkpoint(pilot)

    def malformed(prompt):
        extra = fact(id="extra", question="OTHER")
        if bad == "quote":
            extra["quotation"] = "Never published in the synthetic source."
        elif bad == "location":
            extra["location"] = {
                "kind": "document_ir",
                "block_id": "missing",
                "section_path": [],
                "page": None,
            }
        elif bad == "duplicate_fact":
            extra = fact()
        answer = decision(prompt, "record", facts=[extra])
        if bad == "stale_hash":
            answer["journal_sha256"] = "f" * 64
        return answer

    pilot.queue(malformed)
    commit(pilot)
    result = pilot.run()
    assert result["validated"], result
    assert [f["id"] for f in pilot.read("accepted-journal.json")["facts"]] == ["english"]
    assert [e["turn"] for e in pilot.read("journal-events.json")] == [1, 2, 4]

def test_expert_can_use_more_than_twenty_source_actions_before_network_or_wall_limits(pilot):
    pilot.queue(
        lambda prompt: decision(prompt, action=source_action(url="https://u.edu/admissions/1")),
        elapsed=500,
    )
    for source_step in range(1, 23):

        def observe_then_read(prompt, step=source_step):
            return decision(
                prompt,
                "action",
                facts=[fact(id=f"english_{step}", question="OTHER", source_step=step)],

    assert result["wall_elapsed_ms"] < 900_000


def _sealed_root_fixture(tmp_path):
    tmp_path = tmp_path.resolve(strict=True)
    repo = tmp_path / "repo"
    root = repo / "artifacts/expert-500/phase1-v9-synthetic"
    previous = repo / "artifacts/expert-500/phase1-v8-synthetic"
    root.mkdir(parents=True)
    previous.mkdir()
    workspaces = {name: tmp_path / name for name in ("u", "peer")}
    prior_workspace = tmp_path / "prior-selector"
    for workspace in [*workspaces.values(), prior_workspace]:
        workspace.mkdir()
    (root / "workspace-locations.json").write_text(
        json.dumps({name: str(path) for name, path in workspaces.items()})
    )
    (previous / "workspace-locations.json").write_text(
        json.dumps({"old_case": str(prior_workspace)})
    )
    builder = runpy.run_path(str(PILOT))["sealed_roots_for"]
    return builder, repo, root, previous, workspaces, prior_workspace


def _real_local_without_executing(monkeypatch, workspace, sealed_roots):
    from evaluation.research.expert import blind_cli

    # The real local constructor checks the complete denial set. Executable
    # discovery is mocked; no sandbox, model or university process is run.
    monkeypatch.setattr(blind_cli.shutil, "which", lambda name: f"/mock/{name}")
    return Isolation.local(workspace, sealed_roots)


def test_v9_builder_excludes_current_mapping_and_preserves_every_other_denial(
    tmp_path, monkeypatch
):
    builder, repo, root, previous, workspaces, prior = _sealed_root_fixture(tmp_path)
    roots = builder(repo, root, "u", workspaces)
    resolved = {path.resolve() for path in roots}
    assert workspaces["u"] not in resolved
    assert {repo, workspaces["peer"], prior} <= resolved
    assert Path("/Users/wpalish/Documents/Codex").resolve() in resolved
    assert Path("/Users/wpalish/.codex/memories").resolve() in resolved
    assert Path("/Users/wpalish/.codex/sessions").resolve() in resolved
    assert Path("/Users/wpalish/.codex/attachments").resolve() in resolved
    assert Path("/Users/wpalish/Downloads").resolve() in resolved
    assert Path("/tmp/codex-remote-attachments").resolve() in resolved
    assert PILOT.resolve() in resolved
    assert Path("/tmp/unimatch-expert500-journal-pilot-v8.py").resolve() in resolved
    assert Path(__file__).resolve() in resolved
    isolation = _real_local_without_executing(monkeypatch, workspaces["u"], roots)
    assert isolation.workspace == workspaces["u"]
    assert prior in isolation.sealed_roots and workspaces["peer"] in isolation.sealed_roots


def test_v9_builder_uses_canonical_current_mapping_identity_not_root_spelling(tmp_path):
