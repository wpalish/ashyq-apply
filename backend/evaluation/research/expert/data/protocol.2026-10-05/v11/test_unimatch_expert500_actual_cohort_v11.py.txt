def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def read(path: Path) -> Any:
    return json.loads(path.read_text())


def forbidden(*args: Any, **kwargs: Any) -> Any:

@dataclass
class ActualTrial:
    repo: Path
    root: Path
    report: dict[str, Any]
    parent: dict[str, Any]
    manifest: dict[str, Any]
    launcher: dict[str, Any] = field(default_factory=dict)
    assignments: dict[str, str] = field(default_factory=dict)
    receipts: dict[str, dict[str, Any]] = field(default_factory=dict)
    allocated: list[Path] = field(default_factory=list)
    allocation_arguments: list[Any] = field(default_factory=list)
    git_arguments: list[list[str]] = field(default_factory=list)
    redirected_paths: list[str] = field(default_factory=list)

    def replace_contract(self) -> None:
        """Rehash a synthetic contract so semantic guards, not hash drift, reject it."""
        write(self.root / "workspace-locations.json", self.assignments)
        write(self.root / "workspace-receipts.json", self.receipts)
        trial = read(self.root / "trial.json")
        trial["workspace_assignments_sha256"] = sha(self.root / "workspace-locations.json")
        trial["workspace_receipts_sha256"] = sha(self.root / "workspace-receipts.json")
        write(self.root / "trial.json", trial)
        (self.root / "trial-at-baseline-start.json").write_bytes(
            (self.root / "trial.json").read_bytes()
        )

def mutate_contract(fixture: ActualTrial, mutation: str) -> None:
    first, last = next(iter(COHORT)), next(reversed(COHORT))
    if mutation == "missing_case":
        fixture.assignments.pop(last)
        fixture.receipts.pop(last)
    elif mutation == "extra_case":
        fixture.assignments["unexpected"] = fixture.assignments[first]
        fixture.receipts["unexpected"] = dict(fixture.receipts[first])
    elif mutation == "duplicate_workspace":
        fixture.assignments[last] = fixture.assignments[first]
        fixture.receipts[last] = dict(fixture.receipts[first])
    elif mutation in {"dot_segment", "symlink_ancestor", "trailing_slash"}:
        original = Path(fixture.assignments[first])
        if mutation == "dot_segment":
            alias = str(original.parent) + "/./" + original.name
        elif mutation == "trailing_slash":
            alias = str(original) + "/"
        else:
            parent_alias = fixture.repo / "workspace-parent-alias"
            parent_alias.symlink_to(original.parent, target_is_directory=True)
            alias = str(parent_alias / original.name)
        assert Path(alias).samefile(original)
        fixture.assignments[first] = alias
        fixture.receipts[first]["path"] = alias
    elif mutation in {"bool_device", "bool_inode", "wrong_inode"}:
        field = "device" if mutation == "bool_device" else "inode"
        fixture.receipts[first][field] = (
            fixture.receipts[first][field] + 1 if mutation == "wrong_inode" else True
        )
    else:
        raise AssertionError("Unknown synthetic mutation")
    fixture.replace_contract()

@pytest.mark.parametrize("failure", ["exception", "http_429"])
def test_actual_baseline_batch_failure_is_sticky_and_cannot_start_any_expert(
    actual_trial, monkeypatch, failure
):
    fixture = actual_trial
    calls: list[str] = []
    expert_calls: list[str] = []
    trial_bytes = (fixture.root / "trial.json").read_bytes()

    if failure == "exception":

        def failed_baseline(repo: Path, ref: str, root: Path, case_id: str) -> dict[str, Any]:
            assert repo == fixture.repo and ref == REF and root == fixture.root
            calls.append(case_id)
            raise RuntimeError("Synthetic baseline failure; no source dispatched")

        fixture.parent["baseline"] = failed_baseline
    else:
        # Exercise the actual baseline adapter's recorded429 -> STOP transition.
        def completed429(
            repo: Path, ref: str, case_id: str, inputs: dict[str, Any], destination: Path
        ):
            assert repo == fixture.repo and ref == REF and inputs["mode"] == "expert_baseline"
            assert inputs["search_provider"] == "exa"
            calls.append(case_id)
            budget = {
                "stop_reason": "http_429",
                "elapsed_ms": 10,
                "fetcher_calls": 1,
                "search_queries": 1,
            }
            write(
                destination / "artifact/capture/mock/capture.json",
                {
                    "pipeline_sha": HEAD,
                    "config": {
                        "search_provider": "exa",
                        "strict_expert_budget": True,
                        "case_budgets": {case_id: budget},
                    },
                },
            )
            write(
                destination / "artifact/packet.json",
                {
                    "schema_version": "1",
                    "baseline_pipeline_sha": HEAD,
                    "baseline_budget": {
                        "max_fetcher_calls": 60,
                        "max_elapsed_ms": 120000,
                        "browser_enabled": False,
                    },
                    "cases": [],
                },
            )
            return 1, HEAD

        fixture.parent["workflow"] = completed429

    def forbidden_expert(repo, ref, root, case_id, workspaces):
        expert_calls.append(case_id)
        raise AssertionError("Stopped baseline reached an expert callback")

    fixture.parent["expert"] = forbidden_expert
    stage(fixture, monkeypatch, "baseline")
    assert 1 <= len(calls) <= 3 and len(calls) == len(set(calls))
    final = read(fixture.root / "baseline-results.json")
    assert final["stopped"] is True and final["remaining_case_ids"]
    assert fixture.parent["STOP"].is_set()
    if failure == "http_429":
        assert all(item["budget"]["stop_reason"] == "http_429" for item in final["results"])
        assert all(item["provider"] == "exa" for item in final["results"])
        for case_id in calls:
            packet = read(fixture.root / "baseline" / case_id / "public-packet.json")
            assert packet["comparison_mode"] == NETWORK_COMPARISON_MODE
            assert packet["expert_budget"] == expert_budget()
            assert packet["baseline_budget"]["max_elapsed_ms"] == 120000
    # A fresh process event cannot erase the saved provider/failed-stage receipt.
    fixture.parent["STOP"].clear()
    with pytest.raises(ValueError, match="stopped/incomplete baseline"):
        stage(fixture, monkeypatch, "experts")
    with pytest.raises(ValueError, match="reset or retry"):
        stage(fixture, monkeypatch, "baseline")
    assert expert_calls == [] and not (fixture.root / "experts-results.json").exists()
    assert fixture.allocation_arguments == [tuple(COHORT)]
    assert (fixture.root / "trial.json").read_bytes() == trial_bytes

    check = case["checks"][-3]
    assert check["kind"] == "repository"
    path = Path(check["path"])

    if mutation == "wrong_target":

        "schema_bytes",
        "argv_path",
        "argv_effort",
        "peer_output",
        "stream_hash",
        "symlink_copy",
    ],

def test_v11_new_neutral_replays_actual_bytes_not_success_flags(
    synthetic_neutral, tmp_path, monkeypatch, mutation
):
    probe = copy.deepcopy(synthetic_neutral["probe"])
    copies = {}
    paths = {probe[key + "_path"] for key in ("workspace_assignments", "workspace_receipts")}
    for phase in probe["results"]:
        paths.update(
            phase[key + "_path"] for key in ("request", "stdout", "stderr", "schema", "output")
        )
    for index, original in enumerate(sorted(paths)):
        destination = tmp_path / f"copy-{index}.txt"
        destination.write_bytes(Path(original).read_bytes())
        copies[original] = destination
    phase = probe["results"][0]
    stdout, request = copies[phase["stdout_path"]], copies[phase["request_path"]]
    if mutation in {"tool_event", "raw_failure", "incomplete_stream"}:
        events = [json.loads(line) for line in stdout.read_text().splitlines()]
        if mutation == "tool_event":
            events.insert(
                -1,
                {
                    "type": "item.completed",
                    "item": {"type": "command_execution", "command": "invented"},
                },
            )
        elif mutation == "raw_failure":
            events.insert(-1, {"type": "error", "message": "Invented transport failure"})
        else:
            events = events[:-1]
        stdout.write_text("".join(json.dumps(event) + "\n" for event in events))
        phase["stdout_sha256"] = sha(stdout)
    elif mutation == "output_disagreement":
        answer = copy.deepcopy(phase["answer"])
        answer["observation"] = "Different invented atomic output"
        write(copies[phase["output_path"]], answer)
        phase["output_sha256"] = sha(copies[phase["output_path"]])
    elif mutation == "schema_bytes":
        write(copies[phase["schema_path"]], {"type": "integer"})
    elif mutation in {"argv_path", "argv_effort"}:
        changed = read(request)
        if mutation == "argv_path":
            argv = changed["selector_command"]
            argv[argv.index("--output-last-message") + 1] = "/tmp/different-output.json"
        else:
            changed["selector_command"].extend(["-c", 'model_reasoning_effort="low"'])
        write(request, changed)
        phase["request_sha256"] = sha(request)
    elif mutation == "peer_output":
        phase["output_path"] = str(
            Path(probe["workspace_assignments"]["othercase"]) / "selector-01.answer.json"
        )
    elif mutation == "stream_hash":
        phase["stdout_sha256"] = "0" * 64
    else:
        target = copies[phase["output_path"]]
        target.unlink()
        target.symlink_to(stdout)
    monkeypatch.setattr(subprocess, "run", forbidden)
    monkeypatch.setattr(subprocess, "Popen", forbidden)
    with pytest.raises(ValueError):
        synthetic_neutral["freeze"]["validate_neutral"](
            probe, neutral_script=NEUTRAL_SCRIPT, frozen_cli=probe["codex_cli"], archived=copies
        )

def test_v11_fresh_neutral_replays_archived_streams_without_ephemeral_raw_reads(monkeypatch):
    freeze = runpy.run_path(str(FREEZER))
    archived = freeze["validate_neutral_archive"]()
    probe = read(archived[str(NEUTRAL_RESULT)])
    original_read = Path.read_text
    original_bytes = Path.read_bytes
    raw_originals = set(archived) - {str(NEUTRAL_SCRIPT), str(NEUTRAL_RESULT)}

    def no_original_text(path, *args, **kwargs):
        assert str(path) not in raw_originals, "Ephemeral original raw receipt was read"
        return original_read(path, *args, **kwargs)

    def no_original_bytes(path, *args, **kwargs):
        assert str(path) not in raw_originals, "Ephemeral original raw receipt was read"
        return original_bytes(path, *args, **kwargs)

    monkeypatch.setattr(Path, "read_text", no_original_text)
    monkeypatch.setattr(Path, "read_bytes", no_original_bytes)
    monkeypatch.setattr(subprocess, "run", forbidden)
    monkeypatch.setattr(subprocess, "Popen", forbidden)
    freeze["validate_neutral"](
        probe, neutral_script=NEUTRAL_SCRIPT, frozen_cli=probe["codex_cli"], archived=archived
    )
    reference = freeze["validate_neutral_current"]()
    assert reference["original_manifest_sha256"] == freeze["NEUTRAL_MANIFEST_SHA256"]
    assert (
        reference["runtime_modules"]
        == probe["runtime_modules_before"]
        == probe["runtime_modules_after"]
    )
