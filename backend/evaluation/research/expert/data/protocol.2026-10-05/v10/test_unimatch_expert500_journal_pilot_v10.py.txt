def source_action(tool="fetcher", **changes):
    value = {
        "tool": tool,
        "url": URL if tool == "fetcher" else None,
        "query": "site:u.edu official admissions" if tool == "search_provider" else None,
        "domains": ["u.edu"] if tool == "search_provider" else [],
        "max_results": 4 if tool == "search_provider" else None,
        "source_step": 1 if tool == "view" else None,
        "page": None,
        "offset": None,
        "section_offset": None,
        "link_offset": None,
    }
    return {**value, **changes}


def fact(**changes):
    value = {
        "id": "english",
        "question": "Q1",
        "claim_key": "ielts.overall",
        "value_json": "6.5",
        "source_step": 1,
        "quotation": QUOTE,
        "location": {"kind": "text", "block_id": None, "section_path": [], "page": None},
        "scope": dict(SCOPE),
    }
    return {**value, **changes}


def finding(**changes):
    value = {
        "question": "Q1",
        "status": "found",
        "fact_ids": ["english"],
        "scope": dict(SCOPE),
        "observation": "The recorded synthetic page states this minimum.",
        "supporting_steps": [],
    }
    return {**value, **changes}


def decision(prompt, kind="action", facts=None, findings=None, action=None, **changes):
    if "accepted_journal" not in prompt:
        assert kind == "submit" and action is None and not facts and not findings
        return minimal_commit(prompt, **changes)
    value = {
        "kind": kind,
        "observation": "Inspect the current recorded official source.",
        "journal_sha256": prompt["accepted_journal"]["sha256"],
        "action": source_action() if kind == "action" and action is None else action,
        "facts": facts or [],
        "question_findings": findings or [],
    }
    return {**value, **changes}


def minimal_commit(prompt, **changes):
    from evaluation.research.expert.representation import CLOSING_OBSERVATION

    current_hash = (
        prompt["accepted_journal"]["sha256"]
        if "accepted_journal" in prompt
        else prompt["journal_sha256"]
    )
    return {
        "kind": "submit",
        "observation": CLOSING_OBSERVATION,
        "journal_sha256": current_hash,
        "action": None,
        "facts": [],
        "question_findings": [],
        **changes,
    }


def completed_events(answer):
    return (
        "\n".join(
            json.dumps(item)
            for item in [
                {"type": "thread.started", "thread_id": "synthetic"},
                {"type": "turn.started"},
                {
                    "type": "item.completed",
                    "item": {"id": "answer", "type": "agent_message", "text": answer},
                },
                {"type": "turn.completed", "usage": {"input_tokens": 1, "output_tokens": 1}},
            ]
        )
        + "\n"
    )


@dataclass
class Harness:
    runtime: dict
    repo: Path
    root: Path
    workspace: Path
    packet: dict
    pending: list = field(default_factory=list)
    calls: list = field(default_factory=list)
    queries: list = field(default_factory=list)
    prompts: list = field(default_factory=list)
    commands: list = field(default_factory=list)
    plans: list = field(default_factory=list)
    controller_dirs: list = field(default_factory=list)
    summary_hook: object = None
    workflow_error: Exception | None = None
    source_body: bytes | None = None
    source_content_type: str = "text/html"
    plan_hook: object = None
    search_error_status: int = 429

    @property
    def folder(self):
        return self.root / "experts/u"

    def queue(
        self,
        callback=None,
        *,
        elapsed=12000,
        preparation=0,
        error=None,
        raw=None,
        failed_stream=None,
    ):
        self.pending.append(
            {
                "callback": callback,
                "elapsed": elapsed,
                "preparation": preparation,
                "error": error,
                "raw": raw,
                "failed_stream": failed_stream,
            }
        )

    def read(self, name):
        return json.loads((self.folder / name).read_text())

    def run(self):
        return self.runtime["expert"](
            self.repo, "synthetic-frozen", self.root, "u", {"u": self.workspace}
        )

def bootstrap(pilot):
    pilot.queue()


def checkpoint(pilot, *, kind="record", elapsed=12000, **changes):
    pilot.queue(
        lambda prompt: decision(prompt, kind, facts=[fact()], findings=[finding()], **changes),
        elapsed=elapsed,
    )


def commit(pilot, *, elapsed=500, acknowledge_source=None):
    pilot.queue(
        lambda prompt: decision(
            prompt,
            "submit",
            facts=[
                fact(id="english_second_source", question="OTHER", source_step=acknowledge_source)
            ]
            if acknowledge_source is not None
            else [],
        ),
        elapsed=elapsed,
    )


def assert_exact_clock(result):
    cumulative = 0
    for entry in result["meter"]:
        assert type(entry["elapsed_ms"]) is int and entry["elapsed_ms"] >= 0
        cumulative += entry["elapsed_ms"]
        assert entry["active_ms"] == cumulative
    assert result["active_ms"] == cumulative


class FakeDatetime(datetime):
    current = datetime.now(UTC)

    @classmethod
    def now(cls, tz=None):
        return cls.current.astimezone(tz or UTC)

def test_shared_model_stop_keeps_running_receipts_and_never_starts_queued_seven(

):
    import hashlib

    module = runpy.run_path(str(PILOT))
    runtime = module["main"].__globals__
    runtime["STOP"].clear()

    root.mkdir()

        "protocol_version": "journal-v10",
        "frozen_ref": "synthetic",
        "algorithm_provider": "exa",
        "expert_provider": "exa",
        "search_provider": "exa",
        "workspace_assignments_sha256": hashlib.sha256(
            (root / "workspace-locations.json").read_bytes()
        ).hexdigest(),
        "workspace_receipts_sha256": hashlib.sha256(
            (root / "workspace-receipts.json").read_bytes()
        ).hexdigest(),
    }
    (root / "trial.json").write_text(json.dumps(trial))
    (root / "baseline-results.json").write_text(
        json.dumps(
            {
                "stopped": False,
                "remaining_case_ids": [],
                "results": [
                    {
                        "case_id": c,
                        "provider": "exa",
                        "provider_failure": None,
                        "budget": {"provider_failure": None},
                    }
                    for c in cohort
                ],
            }
        )
    )
    started, lock, barrier = [], threading.Lock(), threading.Barrier(3)

    def expert(repo, ref, result_root, case_id, workspaces):
        with lock:
            started.append(case_id)
        barrier.wait(timeout=10)
        if case_id == cohort[0]:
            result = {
                "case_id": case_id,
                "status": "incomplete",
                "model_failure": {"kind": "model_usage_limit"},
            }
        else:
            assert runtime["STOP"].wait(timeout=10), "Parent never propagated model stop"
            result = {"case_id": case_id, "status": "incomplete", "error": "batch_stopped"}
        folder = result_root / "experts" / case_id
        folder.mkdir(parents=True)
        (folder / "original-running-receipt.json").write_text(json.dumps(result))
        return result

    monkeypatch.setitem(runtime, "expert", expert)
    monkeypatch.setattr(
        sys,
        "argv",
        [
            str(PILOT),
            "--repo",
            str(tmp_path),
            "--root",
            str(root),
            "--ref",
            "synthetic",
            "--stage",
            "experts",
        ],
    )
    runtime["main"]()
    results = json.loads((root / "experts-results.json").read_text())
    assert results["stopped"] is True
    assert set(started) == set(cohort[:3]) and len(started) == 3
    assert set(results["remaining_case_ids"]) == set(cohort[3:])
    assert len(results["results"]) == 3
    assert len(list((root / "experts").glob("*/original-running-receipt.json"))) == 3
    assert not any((root / "experts" / case).exists() for case in cohort[3:])

@pytest.mark.parametrize("bad", ["quote", "location", "stale_hash", "duplicate_fact"])
def test_network_mode_does_not_weaken_atomic_source_citation_guards(pilot, bad):
    bootstrap(pilot)
    checkpoint(pilot)

    def malformed(prompt):
        extra = fact(id="extra", question="OTHER")
        if bad == "quote":
            extra["quotation"] = "Never published in the synthetic source."
        elif bad == "location":
            extra["location"] = {
                "kind": "document_ir",
                "block_id": "missing",
                "section_path": [],
                "page": None,
            }
        elif bad == "duplicate_fact":
            extra = fact()
        answer = decision(prompt, "record", facts=[extra])
        if bad == "stale_hash":
            answer["journal_sha256"] = "f" * 64
        return answer

    pilot.queue(malformed)
    commit(pilot)
    result = pilot.run()
    assert result["validated"], result
    assert [f["id"] for f in pilot.read("accepted-journal.json")["facts"]] == ["english"]

@pytest.mark.parametrize("stage", ["baseline", "experts"])
def test_v10_concurrent_parent_cannot_start_duplicate_stage_before_first_completion(
    tmp_path, monkeypatch, stage
):
    runtime, root, _ = _once_only_stage(tmp_path, monkeypatch, stage)
    entered, release = threading.Event(), threading.Event()
    calls, errors = [], {}

    def controlled_callback(*args):
        calls.append(args[3])
        entered.set()
        assert release.wait(timeout=10)
        return {"case_id": args[3], "provider": "exa"}

    def run_parent(name):
        try:
            runtime["main"]()
        except Exception as error:
            errors[name] = error

    monkeypatch.setitem(
        runtime, "baseline" if stage == "baseline" else "expert", controlled_callback
    )
    first = threading.Thread(target=run_parent, args=("first",))
    second = threading.Thread(target=run_parent, args=("second",))
    first.start()
    try:
        assert entered.wait(timeout=10)
        assert not (root / f"{stage}-results.json").exists()
        second.start()
        second.join(timeout=2)
        assert not second.is_alive(), "Second parent queued duplicate work before first completion"
        assert isinstance(errors.get("second"), ValueError)
        assert calls == ["u"] and not runtime["STOP"].is_set()
    finally:
        release.set()
        first.join(timeout=10)
        if second.ident is not None:
            second.join(timeout=10)
    assert not first.is_alive() and "first" not in errors
    assert calls == ["u"]

    original = b'{"original_dispatch":true}\n'
    if kind == "dangling_symlink":
        destination.symlink_to(tmp_path / "absent")
    else:
        destination.mkdir()
        if kind == "dispatch":
            (destination / "dispatch.json").write_bytes(original)
    calls = []
    monkeypatch.setitem(runtime["legacy"], "workflow", lambda *args: calls.append("GH_DISPATCH"))
    with pytest.raises(RuntimeError, match="cannot reuse"):
        runtime["workflow"](
            tmp_path,
            "synthetic",
            "u",
            {"mode": "expert_baseline", "search_provider": "exa"},
            destination,
        )
    assert calls == []
    if kind == "dispatch":
        assert (destination / "dispatch.json").read_bytes() == original
    elif kind == "dangling_symlink":
        assert destination.is_symlink() and not destination.exists()


def test_v10_existing_expert_destination_refuses_before_new_guard_or_model(pilot):
    pilot.folder.mkdir(parents=True)
    original = b'{"original_case_wall_started_at":"synthetic"}\n'
    receipt = pilot.folder / "original-case.json"
    receipt.write_bytes(original)
    pilot.queue()
    with pytest.raises(RuntimeError, match="cannot reuse"):
        pilot.run()
    assert receipt.read_bytes() == original
    assert not pilot.prompts and not pilot.calls and not pilot.queries and not pilot.plans
    assert not (pilot.workspace / "public-sentinel.txt").exists()
