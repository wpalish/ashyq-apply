PILOT = Path("/tmp/unimatch-expert500-journal-pilot-v11.py")
START = datetime(2026, 10, 4, tzinfo=UTC)
HEAD = "a" * 40
RUN = 123


@dataclass
class Clock:
    seconds: float = 100.0

    def now(self, tz: Any = None) -> datetime:
        assert tz is UTC
        return START + timedelta(seconds=self.seconds)

    def advance(self, seconds: float) -> None:
        self.seconds += seconds


def save(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2) + "\n")


@dataclass
class Harness:
    root: Path
    clock: Clock = field(default_factory=Clock)
    stop: threading.Event = field(default_factory=threading.Event)
    calls: list[dict[str, Any]] = field(default_factory=list)
    dispatch_expiry: bool = False
    ambiguous_dispatch: bool = False
    known_expiry: bool = False
    unidentified: bool = False
    artifact_kind: str = "valid"
    source_kind: str = "success"
    provider_status: int = 503
    cancelled: bool = False
    cancel_error: bool = False
    identity_change: bool = False
    duplicate_tag: bool = False
    invalid_identity: bool = False
    audit_overrun: bool = False
    partial_download: bool = False
    receipt_expiry: bool = False
    submitted: dict[str, Any] = field(default_factory=dict)
    generated: dict[str, Any] = field(default_factory=dict)

    @property
    def guard(self) -> WallGuard:
        return WallGuard(START, START + timedelta(milliseconds=WALL_GUARD_MS))

    @property
    def destination(self) -> Path:
        return self.root / "controller-01"

    def plan(self) -> dict[str, Any]:
        tool = (
            "search_provider" if self.source_kind in {"http_429", "provider_error"} else "fetcher"
        )
        action = (
            {
                "tool": tool,
                "query": "site:u.edu public admissions",
                "domains": ["u.edu"],
            }
            if tool == "search_provider"
            else {"tool": tool, "url": "https://u.edu/admissions"}
        )
        return {
            "case_id": "u",
            "provider": "exa",
            "log_prefix": "u",
            "strict_budget": True,
            "max_calls": 60,
            "max_ms": WALL_GUARD_MS,
            "max_search_queries": 6,
            "budget_origin": NETWORK_COMPARISON_MODE,
            "budget_model": BUDGET_MODEL,
            **self.guard.as_dict(),
            "official_domains": ["u.edu"],
            "actions": [action],
        }

    def write_artifact(self, folder: Path) -> None:
        folder.mkdir(parents=True, exist_ok=True)
        plan = json.loads(self.submitted["plan"])
        save(folder / "plan.json", plan)
        source_clock = Clock(899.0 if self.source_kind == "timeout" else 10.0)
        source_ms = [0.0]

        async def get(url: str) -> FetchResult:
            if self.source_kind == "timeout":
                source_clock.advance(1.02)
                source_ms[0] += 0.02
                raise TimeoutError
            if self.source_kind == "fetcher_403":
                return FetchResult(
                    url=url,
                    final_url=url,
                    outcome=FetchOutcome.HTTP_ERROR,
                    status_code=403,
                    error="Synthetic university refusal",
                )
            return FetchResult(
                url=url,
                outcome=FetchOutcome.OK,
                final_url=url,
                status_code=200,
                content_type="text/html",
                content=b"<p>Public synthetic source information.</p>",
            )

        async def search(**kwargs: Any) -> Any:
            assert self.source_kind in {"http_429", "provider_error"}
            raise SearchUnavailable(
                "Synthetic provider failure",
                http_status=429 if self.source_kind == "http_429" else self.provider_status,
            )

        summary = asyncio.run(
            execute(
                plan,
                folder / "logs",
                get=get,
                search=search,
                provider_name="exa",
                now=lambda: source_clock.now(UTC),
                monotonic=lambda: source_ms[0],
            )
        )
        save(folder / "summary.json", summary)
        self.generated = summary
        name = summary["actions"][0]["raw_log_path"]
        if self.artifact_kind == "missing":
            (folder / "summary.json").unlink()
        elif self.artifact_kind == "tampered":
            (folder / "logs" / name).write_text("{}")
        elif self.artifact_kind == "extra":
            save(folder / "logs" / "u-002.json", {"executed": True})
        elif self.artifact_kind == "wrong_plan":
            plan["actions"][0]["url"] = "https://u.edu/other"
            save(folder / "plan.json", plan)
        elif self.artifact_kind == "symlink":
            raw = folder / "logs" / name
            elsewhere = folder / "outside.json"
            elsewhere.write_bytes(raw.read_bytes())
            raw.unlink()
            raw.symlink_to(elsewhere)
        elif self.artifact_kind in {"mixed_provider", "provider_error_as_success"}:
            raw_path = folder / "logs" / name
            raw_record = json.loads(raw_path.read_text())
            if self.artifact_kind == "mixed_provider":
                raw_record["provider"] = "exa_mcp"
                summary["provider"] = "exa_mcp"
            else:
                raw_record["outcome_status"] = "success"
                summary["actions"][0]["outcome_status"] = "success"
                summary["actions"][0]["outcome"] = "ok"
            save(raw_path, raw_record)
            sha = hashlib.sha256(raw_path.read_bytes()).hexdigest()
            summary["raw_logs"][name] = sha
            summary["actions"][0]["raw_log_sha256"] = sha
            save(folder / "summary.json", summary)

    def command(
        self,
        argv: list[str],
        *,
        cwd: Path,
        stdin: str | None = None,
        timeout: float = 60,
    ) -> str:
        assert cwd == self.root
        assert 0 < timeout <= 60
        self.calls.append({"argv": argv, "timeout": timeout, "at": self.clock.seconds})
        if argv[:3] == ["gh", "workflow", "run"]:
            assert len([call for call in self.calls if call["argv"][1] == "workflow"]) == 1
            self.submitted = json.loads(stdin or "{}")
            assert self.submitted["search_provider"] == "exa"
            assert (self.destination / "dispatch.json").is_file()
            if self.dispatch_expiry or self.ambiguous_dispatch:
                self.clock.seconds = 900.0
            if self.ambiguous_dispatch:
                raise TimeoutError("Ambiguous source dispatch")
            return ""
        assert argv[:2] == ["gh", "run"] and argv[2] in {
            "list",
            "view",
            "cancel",
            "download",
        }
        if self.audit_overrun:
            self.clock.advance(60.001)
            return "[]"
        if argv[2] == "list":
            if self.unidentified:
                return "[]"
            match = {
                "databaseId": True if self.invalid_identity else RUN,
                "displayTitle": "Synthetic capture " + self.submitted["run_tag"],
                "headSha": HEAD,
                "status": "in_progress",
                "conclusion": None,
            }
            # A different existing run cannot be selected by list ordering.
            result = [{**match, "databaseId": 999, "displayTitle": "other-tag"}, match]
            if self.duplicate_tag:
                result.append({**match, "databaseId": 456})
            return json.dumps(result)
        assert argv[3] == str(RUN)
        if argv[2] == "view":
            if self.known_expiry and self.clock.seconds < 900:
                self.clock.seconds = 900.0
                raise TimeoutError("Known source run expired")
            pending = self.clock.seconds >= 900 and not self.cancelled
            return json.dumps(
                {
                    "headSha": "b" * 40 if self.identity_change else HEAD,
                    "status": "in_progress" if pending else "completed",
                    "conclusion": "cancelled" if self.cancelled else "success",
                    "url": "https://example.invalid/run/123",
                }
            )
        if argv[2] == "cancel":
            self.cancelled = True
            if self.cancel_error:
                raise RuntimeError("Cancel raced with completion")
            return ""
        assert argv[argv.index("--name") + 1] == f"expert-recovery-{RUN}"
        folder = Path(argv[argv.index("--dir") + 1])
        if self.partial_download and folder.name == "artifact":
            folder.mkdir()
            save(folder / "plan.json", json.loads(self.submitted["plan"]))
            self.clock.seconds = 900.0
            raise TimeoutError("Interrupted existing artifact download")
        self.write_artifact(folder)
        return ""

    def workflow(self, **kwargs: Any) -> tuple[int, str]:
        source = PILOT.read_text()
        parsed = ast.parse(source)
        function = next(
            node
            for node in parsed.body
            if isinstance(node, ast.FunctionDef) and node.name == "workflow"
        )
        classes = [
            node
            for node in parsed.body
            if isinstance(node, ast.ClassDef)
            and node.name in {"CaseGuardEnded", "_ArtifactUnavailable", "_TransportIntegrityError"}
        ]
        assert len(classes) == 3
        module = ast.Module(body=[*classes, function], type_ignores=[])

        def saved(path: Path, value: Any) -> None:
            save(path, value)
            if self.receipt_expiry and path.name == "dispatch.json":
                self.clock.seconds = 900.0

        namespace: dict[str, Any] = {
            "Path": Path,
            "Any": Any,
            "WallGuard": WallGuard,
            "STOP": self.stop,
            "legacy": {
                "command": self.command,
                "workflow": lambda *args: (777, HEAD),
            },
            "save": saved,
            "emit": lambda **event: None,
            "datetime": SimpleNamespace(now=self.clock.now),
            "UTC": UTC,
            "time": SimpleNamespace(monotonic=lambda: self.clock.seconds, sleep=self.clock.advance),
            "json": json,
            "math": math,
            "hashlib": hashlib,
        }
        exec(compile(module, str(PILOT), "exec"), namespace)
        return namespace["workflow"](
            self.root,
            "frozen-ref",
            "u",
            kwargs.pop("inputs", {"mode": "expert_actions", "plan": json.dumps(self.plan())}),
            self.destination,
            wall_guard=kwargs.pop("wall_guard", self.guard),
            **kwargs,
        )

    def receipt(self) -> dict[str, Any]:
        return json.loads((self.destination / "transport-audit.json").read_text())

    def assert_single_dispatch(self) -> None:
        assert sum(call["argv"][:3] == ["gh", "workflow", "run"] for call in self.calls) == 1
        postguard = [call for call in self.calls if call["at"] >= 900]
        assert all(
            call["argv"][:2] == ["gh", "run"]
            and call["argv"][2] in {"list", "view", "cancel", "download"}
            for call in postguard
        )


def test_normal_workflow_verifies_original_artifact_without_audit_allowance(
    tmp_path: Path,
) -> None:
    harness = Harness(tmp_path)
    assert harness.workflow() == (RUN, HEAD)
    receipt = harness.receipt()
    assert receipt["status"] == "completed"
    assert receipt["audit_only"] is False and receipt["audit_window_ms"] == 0
    assert receipt["artifact_verified"] is True and receipt["dispatch_count"] == 1
    assert receipt["provider"] == harness.generated["provider"] == "exa"
    assert receipt["source_counters"] == {
        "fetcher_calls": 1,
        "search_queries": 0,
        "elapsed_ms": 0,
    }
    assert (
        receipt["summary_sha256"]
        == hashlib.sha256((harness.destination / "artifact/summary.json").read_bytes()).hexdigest()
    )
    assert not harness.stop.is_set()
    harness.assert_single_dispatch()

@pytest.mark.parametrize("source_kind", ["success", "timeout", "http_429"])
def test_known_expiry_cancels_downloads_and_preserves_original_source_stop(
    tmp_path: Path, source_kind: str
) -> None:
    harness = Harness(tmp_path, known_expiry=True, source_kind=source_kind)
    assert harness.workflow() == (RUN, HEAD)
    receipt = harness.receipt()
    expected = {
        "success": None,
        "timeout": "wall_guard_exhausted",
        "http_429": "http_429",
    }
    assert receipt["source_stop_reason"] == expected[source_kind]
    assert receipt["source_counters"] == {
        "fetcher_calls": int(source_kind != "http_429"),
        "search_queries": int(source_kind == "http_429"),
        "elapsed_ms": 20 if source_kind == "timeout" else 0,
    }
    assert receipt["raw_logs"] == harness.generated["raw_logs"]
    assert harness.stop.is_set() is (source_kind == "http_429")
    harness.assert_single_dispatch()

def test_v11_baseline_legacy_adapter_receives_explicit_exa(tmp_path: Path) -> None:
    source = ast.parse(PILOT.read_text())
    function = next(
        n for n in source.body if isinstance(n, ast.FunctionDef) and n.name == "workflow"
    )
    seen: list[tuple[Any, ...]] = []

    def legacy_workflow(*args: Any) -> tuple[int, str]:
        seen.append(args)
        return 777, HEAD

    namespace: dict[str, Any] = {
        "Path": Path,
        "Any": Any,
        "WallGuard": WallGuard,
        "STOP": threading.Event(),
        "legacy": {"workflow": legacy_workflow},
    }
    exec(compile(ast.Module(body=[function], type_ignores=[]), str(PILOT), "exec"), namespace)
    assert namespace["workflow"](
        tmp_path, "ref", "u", {"mode": "expert_baseline"}, tmp_path / "baseline"
    ) == (777, HEAD)
    assert seen[0][3]["search_provider"] == "exa"
