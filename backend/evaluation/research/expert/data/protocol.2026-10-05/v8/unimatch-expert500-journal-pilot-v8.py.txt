def validate_all_fact_citations(
    decision: Any, actions: list[dict[str, Any]], *, logs_dir: Path, packet: dict[str, Any]
) -> list[dict[str, Any]]:
    # Original V3 byte-frozen guard validates every source/quote/location first.
    proof = validate_fact_citations(decision, actions, logs_dir=logs_dir, packet=packet)
    if any(
        award_value(fact.claim_key, fact.decoded_value()) is not None for fact in decision.facts
    ):
        from evaluation.research.expert.citations import (
            _case,
            _read_source,
            _recorded_actions,
            _scope_context,
        )
        from evaluation.research.expert.models import EvidenceLocation, SourceEvidence

        case = _case(packet)
        recorded = _recorded_actions(actions, case["case_id"])
        contexts, sources = {}, {}
        for fact in decision.facts:
            if award_value(fact.claim_key, fact.decoded_value()) is None:
                continue
            if fact.source_step not in sources:
                sources[fact.source_step] = _read_source(
                    recorded[fact.source_step], recorded, logs_dir.resolve(), case
                )
            source = sources[fact.source_step]
            evidence = SourceEvidence(
                action_step=fact.source_step,
                source_url=source.url,
                excerpt=fact.quotation,
                location=EvidenceLocation.model_validate(fact.location.model_dump()),
            )
            selected = _scope_context(evidence, source)
            if selected.location.kind == "document_ir":
                block = next(
                    b for b in source.document.blocks if b.id == selected.location.block_id
                )
                contexts[fact.id] = " ".join(
                    [
                        *block.section_path,
                        *block.row_headers,
                        *block.column_headers,
                        block.label,
                        block.caption,
                        block.text,
                    ]
                )
            else:
                contexts[fact.id] = selected.excerpt
        subject_proof = {
            item["fact_id"]: item
            for item in validate_award_subjects(decision, contexts_by_fact_id=contexts)
        }
        proof = [
            {
                **item,
                **(
                    {"award_subject_proof": subject_proof[item["fact_id"]]}
                    if item["fact_id"] in subject_proof
                    else {}
                ),
            }
            for item in proof
        ]
    return proof


@dataclass(frozen=True)
class NetworkResearchState:
    recorded_source_count: int
    processed_source_count: int
    source_closed: bool
    mode: str
    phase: str

    def as_dict(self) -> dict[str, Any]:
        return {
            "policy_version": "network_actions_v1",
            "mode": self.mode,
            "phase": self.phase,
            "recorded_source_count": self.recorded_source_count,
            "processed_source_count": self.processed_source_count,
            "pending_source_count": self.recorded_source_count - self.processed_source_count,
            "source_closed": self.source_closed,
            "can_dispatch_source": not self.source_closed,
        }


def research_state(
    *, recorded_source_count: int, processed_source_count: int, source_closed: bool = False
) -> NetworkResearchState:
    if any(type(v) is not int or v < 0 for v in (recorded_source_count, processed_source_count)):
        raise ValueError("Source counts must be exact nonnegative integers")
    if processed_source_count > recorded_source_count or type(source_closed) is not bool:
        raise ValueError("Invalid processed-source count or source closure")
    mode = "research"
    phase = "bootstrap" if recorded_source_count == 0 else "data"
    if source_closed:
        mode = (
            "minimal_commit"
            if recorded_source_count and processed_source_count == recorded_source_count
            else "data_closure"
        )
        phase = "minimal_commit" if mode == "minimal_commit" else "data"
    return NetworkResearchState(
        recorded_source_count, processed_source_count, source_closed, mode, phase
    )


@dataclass(frozen=True)
class NetworkSourceProposal:
    observed_at: str
    wall_guard: WallGuard
    source_closed: bool
    dispatch_allowed: bool
    reason: str

    def as_dict(self) -> dict[str, Any]:
        return {
            "policy_version": BUDGET_MODEL,
            "observed_at": self.observed_at,
            **self.wall_guard.as_dict(),
            "source_closed": self.source_closed,
            "dispatch_allowed": self.dispatch_allowed,
            "status": "dispatch_allowed" if self.dispatch_allowed else "declined",
            "reason": self.reason,
        }


def source_proposal(*, wall_guard: WallGuard, source_closed: bool = False) -> NetworkSourceProposal:
    now = datetime.now(UTC)
    remaining = wall_guard.remaining_seconds(now=now)
    reason = (
        "wall_guard_exhausted"
        if remaining <= 0
        else "source_closed"
        if source_closed
        else "within_wall_guard"
    )
    return NetworkSourceProposal(
        now.isoformat(), wall_guard, source_closed, remaining > 0 and not source_closed, reason
    )


def make_trace(
    case_id: str,
    packet: dict[str, Any],
    started: str,
    actions: list[dict[str, Any]],
    proposals: list[dict[str, Any]],
    findings: list[dict[str, Any]],
    *,
    wall_guard: WallGuard,
    ended_at: str | None = None,
) -> dict[str, Any]:
    return {
        "schema_version": "1",
        "case_id": case_id,
        "packet_sha256": digest(packet),
        "baseline_pipeline_sha": packet["baseline_pipeline_sha"],
        "max_fetcher_calls": 60,
        "max_elapsed_ms": WALL_GUARD_MS,
        "max_search_queries": 6,
        "initial_fetcher_calls": 0,
        "initial_elapsed_ms": 0,
        "initial_search_queries": 0,
        "budget_model": BUDGET_MODEL,
        **wall_guard.as_dict(),
        "started_at": started,
        "ended_at": ended_at or datetime.now(UTC).isoformat(),
        "actions": actions,
        "proposals": proposals,
        "question_findings": findings,
        "review_status": "unreviewed",
    }


def workflow(
    repo: Path,
    ref: str,
    case_id: str,
    inputs: dict[str, str],
    destination: Path,
    *,
    wall_guard: WallGuard | None = None,
) -> tuple[int, str]:
    if inputs.get("search_provider", "exa") != "exa":
        STOP.set()
        raise ValueError("V8 requires its pinned Exa search provider")
    inputs = {**inputs, "search_provider": "exa"}
    if inputs.get("mode") == "expert_baseline":
        return legacy["workflow"](repo, ref, case_id, inputs, destination)
    if wall_guard is None or STOP.is_set():
        raise RuntimeError("Expert source requires its immutable guard and a live batch")
    import uuid

    from evaluation.research.expert.network_report import (
        SourceCounters,
        validate_source_run,
    )

    # A clean expiry must leave no apparent controller run for the raw replay.
    if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
        raise TimeoutError("wall_guard_exhausted")

    plan = json.loads(inputs["plan"])
    if (
        not isinstance(plan, dict)
        or inputs.get("mode") != "expert_actions"
        or plan.get("provider") != "exa"
    ):
        raise RuntimeError("Expert transport requires its original controlled action plan")
    run_id: int | None = None
    head_sha: str | None = None
    state: dict[str, Any] = {}
    dispatch_attempted = False
    audit_began: float | None = None
    audit_started_at = None
    cancel_attempted = False
    cancel_outcome = None
    artifact_proof: dict[str, Any] = {}
    trigger_error = None

    def audit_remaining() -> float:
        assert audit_began is not None
        seconds = 60.0 - (time.monotonic() - audit_began)
        if seconds <= 0:
            raise TimeoutError("transport_audit_exhausted")
        return seconds

    def remaining() -> float:
        if audit_began is not None:
            return audit_remaining()
        seconds = wall_guard.remaining_seconds(now=datetime.now(UTC))
        if seconds <= 0:
            raise TimeoutError("wall_guard_exhausted")
        return seconds

    def gh(argv: list[str], *, stdin: str | None = None) -> str:
        nonlocal dispatch_attempted
        timeout = min(60.0, remaining())
        if audit_began is not None:
            if argv[:2] != ["gh", "run"] or argv[2] not in {
                "list",
                "view",
                "cancel",
                "download",
            }:
                raise RuntimeError("Transport audit cannot dispatch a new source action")
        elif argv[:3] == ["gh", "workflow", "run"]:
            if STOP.is_set():
                raise RuntimeError("Batch stopped before source dispatch")
            timeout = min(timeout, remaining())
            # Once the command is invoked even an exception is an ambiguous
            # dispatch. Only identifying/auditing this tag is permitted next.
            dispatch_attempted = True
        result = legacy["command"](argv, cwd=repo, stdin=stdin, timeout=timeout)
        if audit_began is not None:
            audit_remaining()
        return result

    tag = "e500-" + uuid.uuid4().hex[:16]
    inputs = {
        "runner": "ubuntu-latest",
        "seconds_per_case": "120",
        "max_pages": "60",
        "search_provider": "exa",
        **inputs,
        "case": case_id,
        "run_tag": tag,
    }

    def receipt(status: str, *, error: str | None = None) -> None:
        began = audit_began
        elapsed = math.ceil((time.monotonic() - began) * 1000) if began is not None else 0
        save(
            destination / "transport-audit.json",
            {
                "status": status,
                "case_id": case_id,
                "provider": "exa",
                "ref": ref,
                "run_tag": tag,
                **wall_guard.as_dict(),
                "dispatch_attempted": dispatch_attempted,
                "dispatch_count": int(dispatch_attempted),
                "audit_only": audit_began is not None,
                "audit_window_ms": 60000 if audit_began is not None else 0,
                "audit_started_at": audit_started_at,
                "audit_elapsed_ms": elapsed,
                "guard_exhausted": wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0,
                "trigger_error": trigger_error,
                "error": error,
                "run_id": run_id,
                "head_sha": head_sha,
                "run_status": state.get("status"),
                "conclusion": state.get("conclusion"),
                "cancel_attempted": cancel_attempted,
                "cancel_outcome": cancel_outcome,
                "artifact_verified": bool(artifact_proof),
                **artifact_proof,
            },
        )

    def identify() -> None:
        nonlocal run_id, head_sha
        while run_id is None:
            runs = json.loads(
                gh(
                    [
                        "gh",
                        "run",
                        "list",
                        "--workflow",
                        "benchmark-capture.yml",
                        "--branch",
                        ref,
                        "--event",
                        "workflow_dispatch",
                        "--limit",
                        "40",
                        "--json",
                        "databaseId,displayTitle,headSha,status,conclusion",
                    ]
                )
            )
            matches = [r for r in runs if r["displayTitle"].endswith(tag)]
            if len(matches) > 1:
                raise RuntimeError("Dispatched tag identifies more than one workflow")
            if matches:
                candidate_id, candidate_sha = matches[0]["databaseId"], matches[0]["headSha"]
                if (
                    type(candidate_id) is not int
                    or candidate_id <= 0
                    or type(candidate_sha) is not str
                    or len(candidate_sha) != 40
                    or any(c not in "0123456789abcdef" for c in candidate_sha)
                ):
                    raise RuntimeError("Dispatched workflow identity is invalid")
                run_id, head_sha = candidate_id, candidate_sha
                save(
                    destination / "run.json",
                    {"run_id": run_id, "head_sha": head_sha, "tag": tag},
                )
                emit(
                    case=case_id,
                    mode=inputs["mode"],
                    run_id=run_id,
                    state="dispatched",
                    source=head_sha,
                )
                return
            if audit_began is None and STOP.is_set():
                raise RuntimeError("Batch stopped before dispatched run could be identified")
            time.sleep(min(5.0, remaining()))

    def completed() -> None:
        nonlocal state, cancel_attempted, cancel_outcome
        while True:
            state = json.loads(
                gh(
                    [
                        "gh",
                        "run",
                        "view",
                        str(run_id),
                        "--json",
                        "status,conclusion,headSha,url",
                    ]
                )
            )
            if state["headSha"] != head_sha:
                raise RuntimeError("Workflow source changed unexpectedly")
            if state["status"] == "completed":
                save(destination / "run.json", {"run_id": run_id, "tag": tag, **state})
                return
            if audit_began is not None and not cancel_attempted:
                cancel_attempted = True
                try:
                    gh(["gh", "run", "cancel", str(run_id)])
                    cancel_outcome = "requested"
                except Exception as error:
                    # A completion/cancel race can still yield a complete
                    # original artifact; never retry this cancellation.
                    cancel_outcome = type(error).__name__
                continue
            if audit_began is None and STOP.is_set():
                raise RuntimeError("Canceled after batch provider429")
            time.sleep(min(5.0 if audit_began is not None else 10.0, remaining()))

    def verify_artifact(folder: Path) -> dict[str, Any]:
        remaining()
        if folder.is_symlink() or not folder.is_dir():
            raise RuntimeError("Original controlled artifact is unavailable")
        logs = folder / "logs"
        if logs.is_symlink() or not logs.is_dir():
            raise RuntimeError("Original controlled raw logs are unavailable")

        def original(path: Path) -> bytes:
            if path.is_symlink() or not path.is_file():
                raise RuntimeError("Original controlled artifact must contain regular files")
            return path.read_bytes()

        if json.loads(original(folder / "plan.json")) != plan:
            raise RuntimeError("Original artifact changed the dispatched action plan")
        summary_bytes = original(folder / "summary.json")
        summary = json.loads(summary_bytes)
        produced, raw_logs = summary.get("actions"), summary.get("raw_logs")
        if not isinstance(produced, list) or len(produced) != 1 or not isinstance(raw_logs, dict):
            raise RuntimeError("Original controller action/checkpoint is incomplete")
        if {path.name for path in logs.iterdir()} != set(raw_logs):
            raise RuntimeError("Original artifact has unaccounted or missing raw source logs")
        records = {}
        provider_failures = []
        for name, expected_sha in raw_logs.items():
            if (
                type(name) is not str
                or name in {"", ".", ".."}
                or Path(name).name != name
                or "/" in name
                or "\\" in name
            ):
                raise RuntimeError("Original raw source paths must be confined")
            data = original(logs / name)
            if hashlib.sha256(data).hexdigest() != expected_sha:
                raise RuntimeError("Original controlled raw source hash changed")
            records[name] = json.loads(data)
            raw = records[name]
            if raw.get("provider") != "exa" or raw.get("case_id") != case_id:
                raise RuntimeError("Original raw source provider/case identity changed")
            if (
                raw.get("executed") is True
                and raw.get("request", {}).get("tool") == "search_provider"
            ):
                response = raw.get("response", {})
                statuses = {
                    response.get(key)
                    for key in ("http_status", "status_code")
                    if type(response.get(key)) is int and response[key] in {401, 402, 403, 429}
                }
                for status in sorted(statuses):
                    provider_failures.append(
                        {
                            "provider": "exa",
                            "http_status": status,
                            "raw_log_path": name,
                            "raw_log_sha256": expected_sha,
                        }
                    )
        name = produced[0]["raw_log_path"]
        resume = plan.get("resume_summary")
        prior = (
            SourceCounters(
                resume["final"]["fetcher_calls"],
                resume["final"]["search_queries"],
                resume["final"]["elapsed_ms"],
            )
            if resume
            else SourceCounters()
        )
        counters = validate_source_run(
            plan,
            summary,
            records[name],
            case_id=case_id,
            official_domains=plan["official_domains"],
            wall_guard=wall_guard,
            raw_sha256=raw_logs[name],
            prior=prior,
            prior_summary=resume,
            provider="exa",
        )
        remaining()
        if provider_failures:
            STOP.set()
        return {
            "summary_sha256": hashlib.sha256(summary_bytes).hexdigest(),
            "plan_sha256": hashlib.sha256(original(folder / "plan.json")).hexdigest(),
            "raw_logs": raw_logs,
            "source_counters": counters.as_dict(),
            "source_stop_reason": summary["final"].get("stop_reason"),
            "provider_failures": provider_failures,
        }

    def artifact() -> None:
        nonlocal artifact_proof
        folder = destination / "artifact"
        if folder.exists():
            try:
                artifact_proof = verify_artifact(folder)
                return
            except Exception:
                if audit_began is None:
                    raise
        download = destination / "artifact-audit" if folder.exists() else folder
        gh(
            [
                "gh",
                "run",
                "download",
                str(run_id),
                "--name",
                f"expert-recovery-{run_id}",
                "--dir",
                str(download),
            ]
        )
        artifact_proof = verify_artifact(download)
        if download != folder:
            folder.rename(destination / "artifact-incomplete")
            download.rename(folder)
        remaining()

    if destination.exists():
        STOP.set()
        raise RuntimeError("Source transport cannot reuse a controller destination")
    destination.mkdir(parents=True, exist_ok=True)
    save(
        destination / "dispatch.json",
        {"ref": ref, "inputs": inputs, **wall_guard.as_dict()},
    )
    try:
        gh(
            ["gh", "workflow", "run", "benchmark-capture.yml", "--ref", ref, "--json"],
            stdin=json.dumps(inputs),
        )
        identify()
        completed()
        artifact()
        if run_id is None or head_sha is None:
            raise RuntimeError("Original dispatched workflow identity is unavailable")
        receipt("completed")
        return run_id, head_sha
    except Exception as error:
        trigger_error = type(error).__name__
        if not dispatch_attempted:
            STOP.set()
            receipt("incomplete", error=trigger_error)
            raise RuntimeError(
                "Source receipt exists but no dispatch outcome is auditable"
            ) from error
        audit_began = time.monotonic()
        audit_started_at = datetime.now(UTC).isoformat()
        try:
            identify()
            completed()
            artifact()
            if run_id is None or head_sha is None:
                raise RuntimeError("Original dispatched workflow identity is unavailable")
            receipt("recovered")
            audit_remaining()
            return run_id, head_sha
        except Exception as audit_error:
            STOP.set()
            receipt("incomplete", error=type(audit_error).__name__)
            raise RuntimeError(
                "Source dispatch transport audit is incomplete; batch stopped"
            ) from audit_error


def baseline(repo: Path, ref: str, root: Path, case_id: str) -> dict[str, Any]:
    case_root = root / "baseline" / case_id
    run_id, head_sha = workflow(
        repo,
        ref,
        case_id,
        {
            "mode": "expert_baseline",
            "search_provider": "exa",
            "extra_args": "--strict-expert-budget --finalization-reserve-ms 1000",
        },
        case_root,
    )
    artifacts = case_root / "artifact"
    captures = list(artifacts.glob("capture/**/capture.json"))
    if len(captures) != 1:
        raise RuntimeError(
            f"Expected one independently saved baseline capture, found {len(captures)}"
        )
    capture = json.loads(captures[0].read_text())
    if capture["pipeline_sha"] != head_sha or capture["config"]["search_provider"] != "exa":
        raise RuntimeError("Baseline revision/provider mismatch")
    if not capture["config"].get("strict_expert_budget"):
        raise RuntimeError("Baseline did not use strict equal-budget mode")
    budget = capture["config"]["case_budgets"][case_id]
    provider_failure = budget.get("provider_failure")
    if provider_failure is not None or budget.get("stop_reason") == "http_429":
        STOP.set()
    packet = json.loads((artifacts / "packet.json").read_text())
    packet["public_profile"] = dict(legacy["PUBLIC_PROFILE"])
    packet["comparison_mode"] = NETWORK_COMPARISON_MODE
    packet["expert_budget"] = expert_budget()
    packet["baseline_budget"]["max_search_queries"] = 6
    save(case_root / "public-packet.json", packet)
    result = {
        "case_id": case_id,
        "provider": "exa",
        "run_id": run_id,
        "head_sha": head_sha,
        "capture": str(captures[0]),
        "packet": str(case_root / "public-packet.json"),
        "budget": budget,
        "provider_failure": provider_failure,
    }
    save(case_root / "result.json", result)
    return result


def ledger_prompt(journal: AcceptedJournal) -> dict[str, Any]:
    submitted = journal.as_submission()
    return {
        "sha256": journal.sha256,
        "facts": [
            {
                "id": f.id,
                "question": f.question,
                "claim_key": f.claim_key,
                "value_json": f.value_json,
                "scope": f.scope.model_dump(),
                "source_step": f.source_step,
            }
            for f in submitted.facts
        ],
        "settled_findings": [
            {
                "question": f.question,
                "status": f.status,
                "fact_ids": f.fact_ids,
            }
            for f in submitted.question_findings
        ],
    }


def selector_packet(packet: dict[str, Any]) -> dict[str, Any]:
    """Identity/request/caps only; freshly measured baseline outputs are not hints."""
    return {
        "schema_version": packet["schema_version"],
        "baseline_budget": packet["baseline_budget"],
        "public_profile": packet.get("public_profile", {}),
        "comparison_mode": NETWORK_COMPARISON_MODE,
        "expert_budget": packet["expert_budget"],
        "cases": [
            {key: case[key] for key in ("case_id", "university", "domain", "request")}
            for case in packet["cases"]
        ],
    }


def commit_events(path: Path, events: list[dict[str, Any]]) -> None:
    """Keep the previous marker intact if writing its replacement is interrupted."""
    pending = path.with_name("." + path.name + ".pending")
    save(pending, events)
    pending.replace(path)


def expert(
    repo: Path, ref: str, root: Path, case_id: str, workspaces: dict[str, Path]
) -> dict[str, Any]:
    started_dt = datetime.now(UTC)
    wall_guard = WallGuard(started_dt, started_dt + timedelta(milliseconds=WALL_GUARD_MS))
    started = started_dt.isoformat()
    case_root = root / "experts" / case_id
    case_root.mkdir(parents=True, exist_ok=True)
    assignments = json.loads((root / "workspace-locations.json").read_text())
    receipts = json.loads((root / "workspace-receipts.json").read_text())
    verified_workspaces = validate_workspaces(assignments, receipts, case_ids=tuple(workspaces))
    if verified_workspaces != workspaces:
        raise ValueError("Caller workspace mapping differs from original canonical assignments")
    workspace = verified_workspaces[case_id]
    sealed_roots = [
        Path("/Users/wpalish/Documents/Codex"),
        Path("/Users/wpalish/.codex/memories"),
        Path("/Users/wpalish/.codex/sessions"),
        Path("/Users/wpalish/.codex/attachments"),
        Path("/Users/wpalish/Downloads"),
        Path("/tmp/codex-remote-attachments"),
    ]
    sealed_roots += [p for name, p in workspaces.items() if name != case_id]
    # External parent scripts/tests/probes may contain nonpublic prior-run context.
    sealed_roots += list(Path("/tmp").glob("unimatch-expert500*"))
    sealed_roots += list(Path("/tmp").glob("test_unimatch_expert500*"))
    for old in (repo / "artifacts/expert-500").glob("phase1*/workspace-locations.json"):
        if old.is_file():
            sealed_roots += [Path(p) for p in json.loads(old.read_text()).values()]
    isolation = replace(Isolation.local(workspace, sealed_roots), codex_executable=CLI)
    sentinel = workspace / "public-sentinel.txt"
    sentinel.write_text("PUBLIC_PACKET_SENTINEL")
    isolation_proof = isolation.prove_access(sentinel, root / "sealed-sentinel.txt")
    isolation_proof.update(
        workspace=str(isolation.workspace),
        sandbox_executable=isolation.sandbox_executable,
        codex_executable=isolation.codex_executable,
        profile=isolation.profile(),
    )
    save(case_root / "isolation.json", isolation_proof)
    schema = workspace / "decision.schema.json"
    baseline_result = json.loads((root / "baseline" / case_id / "result.json").read_text())
    if (
        baseline_result.get("provider") != "exa"
        or baseline_result.get("provider_failure") is not None
        or baseline_result.get("budget", {}).get("provider_failure") is not None
    ):
        STOP.set()
        raise RuntimeError("V8 expert requires an Exa baseline without provider refusal")
    packet = json.loads(Path(baseline_result["packet"]).read_text())
    case = packet["cases"][0]
    context = (repo / "backend/evaluation/research/expert/ALGORITHM_CONTEXT.md").read_text()
    validate_expert_budget(packet)
    active_ms = 0
    source_ms = 0
    accepted_trace_ended_at = started
    meter: list[dict[str, Any]] = []
    history: list[dict[str, Any]] = []
    actions: list[dict[str, Any]] = []
    journal = AcceptedJournal.initial()
    events: list[dict[str, Any]] = []
    summary: dict[str, Any] | None = None
    prior_run: int | None = None
    log_dir: Path | None = None
    feedback: dict[str, Any] | None = None
    closing_only = False
    explicit_commit = False
    navigation_outcomes: list[dict[str, Any]] = []
    turn_errors: list[dict[str, Any]] = []
    result: dict[str, Any] = {"case_id": case_id, "status": "incomplete"}

    def charge(stage: str, operation: Callable[[], T]) -> T:
        nonlocal active_ms
        began = time.monotonic()
        try:
            return operation()
        finally:
            elapsed = math.ceil((time.monotonic() - began) * 1000)
            active_ms += elapsed
            meter.append({"stage": stage, "elapsed_ms": elapsed, "active_ms": active_ms})

    charge(
        "navigation_outcomes_initialization",
        lambda: save(case_root / "navigation-outcomes.json", navigation_outcomes),
    )

    turn = 0
    while True:
        turn += 1
        if STOP.is_set():
            result["error"] = "batch_stopped"
            break
        if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
            result["completion_kind"] = "wall_guard_checkpoint"
            result["error"] = "wall_guard_exhausted"
            break
        if summary and summary["final"].get("stop_reason"):
            closing_only = True
        processed_through = events[-1]["source_action_count"] if events else 0
        state = charge(
            "research_state_policy",
            lambda: research_state(
                recorded_source_count=len(actions),
                processed_source_count=processed_through,
                source_closed=closing_only,
            ),
        )
        meter[-1]["turn"] = turn
        selector_mode, selector_phase = state.mode, state.phase
        closing_only = selector_mode != "research"
        minimal_commit = selector_mode == "minimal_commit"

        def build_prompt() -> dict[str, Any]:
            if minimal_commit:
                return closing_prompt(journal_sha256=journal.sha256)
            return {
                "task": TASK + "\n" + EFFECTIVE_JOURNAL_INSTRUCTIONS,
                "public_packet": selector_packet(packet),
                "algorithm_context": context,
                "public_representation": representation_card(case["university"]),
                "public_turn_policy": policy_card(),
                "public_network_budget": expert_budget(),
                "wall_guard": wall_guard.as_dict(),
                "public_question_contract": question_contract(packet),
                "public_claim_keys": CLAIM_KEYS,
                "public_structured_keys": STRUCTURED_KEYS,
                "public_award_dimensions": AWARD_KEYS,
                "accepted_journal": ledger_prompt(journal),
                "recorded_actions": [
                    {
                        "step": a["step"],
                        "tool": a["tool"],
                        "url": a.get("final_url") or a.get("url"),
                        "outcome": a["outcome_status"],
                    }
                    for a in actions
                ],
                "current_source_views": history[-2:],
                "validation_feedback": feedback,
                "processed_through_action_count": processed_through,
                "required_now": (
                    "Inspect the latest recorded source views and explicitly submit "
                    "newly supported facts or an explicit unresolved/inaccessible question finding. "
                    "An empty hash-only submit cannot acknowledge a new source; no next action."
                    if closing_only
                    else "submit newly observed facts with one next action, or commit"
                ),
                "wall_guard_remaining_ms": wall_guard.remaining_ms(now=datetime.now(UTC)),
                "network_reads_used": summary["final"]["fetcher_calls"] if summary else 0,
                "network_queries_used": summary["final"]["search_queries"] if summary else 0,
                "model_thinking_included_in_network_budget": False,
                "read_limit": 60,
                "query_limit": 6,
            }

        prompt = charge("prompt_construction", build_prompt)
        bound_schema = charge(
            "bound_schema_construction",
            lambda: journal_schema_for_turn(
                journal_sha256=journal.sha256,
                canonical_university=case["university"],
                phase=selector_phase,
            ),
        )
        if closing_only and not minimal_commit:
            bound_schema["properties"]["kind"]["enum"] = ["record", "submit"]
        schema, output = charge(
            "selector_workspace_paths",
            lambda: selector_paths(
                assignments, receipts, case_id=case_id, turn=turn, fresh_output=True
            ),
        )
        charge("bound_schema_write", lambda: save(schema, bound_schema))
        selector_command = charge(
            "selector_command_construction",
            lambda: isolation.command(schema, output=output, model=MODEL),
        )
        command_metadata = charge("selector_command_proof", lambda: command_proof(selector_command))
        request_record = {
            "selector_phase": selector_phase,
            "selector_policy_active_ms": active_ms,
            "turn_policy": policy_limits(),
            "network_budget": expert_budget(),
            **wall_guard.as_dict(),
            "research_state": state.as_dict(),
            "schema_sha256": digest(bound_schema),
            "selector_command": selector_command,
            "command_proof": command_metadata,
            "selector_mode": selector_mode,
            "source_action_count": len(actions),
            "processed_through_action_count": processed_through,
            "prompt": prompt,
            "schema": bound_schema,
        }
        charge(
            "selector_request_write",
            lambda: save(case_root / f"selector-{turn:02d}.request.json", request_record),
        )
        selector_start_active_ms = active_ms
        window = turn_window(phase=selector_phase, wall_guard=wall_guard, now=datetime.now(UTC))
        if window.timeout_ms is None:
            result["completion_kind"] = "wall_guard_checkpoint"
            result["error"] = "wall_guard_exhausted"
            break
        selected = select(
            isolation,
            schema,
            output,
            prompt,
            wall_guard=wall_guard,
            phase=selector_phase,
            model=MODEL,
        )
        window = selected.window
        active_ms += selected.elapsed_ms
        meter.append(
            {
                "stage": "closing_selector" if closing_only else "checkpoint_selector",
                "turn": turn,
                "elapsed_ms": selected.elapsed_ms,
                "active_ms": active_ms,
            }
        )
        # These are audit copies of already captured streams, not source processing.
        (case_root / f"selector-{turn:02d}.events.jsonl").write_text(selected.stdout)
        (case_root / f"selector-{turn:02d}.stderr.txt").write_text(selected.stderr)
        save(
            case_root / f"selector-{turn:02d}.meta.json",
            {
                "selector_phase": selector_phase,
                "selector_start_active_ms": selector_start_active_ms,
                "selector_timeout_ms": window.timeout_ms,
                "selector_timeout_seconds": window.timeout_seconds,
                "selector_window": window.as_dict(),
                "reasoning_effort": None,
                "reasoning_effort_source": "cli_default",
                "schema_sha256": digest(bound_schema),
                "selector_command_sha256": command_metadata["argv_sha256"],
                **wall_guard.as_dict(),
                "model_started_at": selected.selector_started_at,
                "model_ended_at": selected.selector_finished_at,
                "returncode": selected.returncode,
                "error_kind": selected.error_kind,
                "elapsed_ms": selected.elapsed_ms,
            },
        )
        try:
            charge(
                "selector_return_workspace_paths",
                lambda: validate_selector_paths(
                    assignments,
                    receipts,
                    case_id=case_id,
                    turn=turn,
                    workspace=workspace,
                    schema=schema,
                    output=output,
                ),
            )
        except ValueError as exc:
            result["error"] = "selector_workspace_changed: " + str(exc)
            break
        if selected.error_kind:
            turn_errors.append(
                {"turn": turn, "kind": selected.error_kind, "elapsed_ms": selected.elapsed_ms}
            )
            result["error"] = selected.error_kind
            break  # Earlier accepted explicit checkpoints survive; partial text does not.
        model_started = datetime.fromisoformat(selected.selector_started_at)
        model_finished = datetime.fromisoformat(selected.selector_finished_at)
        model_elapsed_ms = math.ceil((model_finished - model_started).total_seconds() * 1000)
        if model_elapsed_ms < 0 or model_elapsed_ms > window.timeout_ms:
            result["error"] = "selector_phase_timeout_exceeded"
            break
        if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
            result["error"] = "selector_completed_after_checkpoint_deadline"
            break
        chosen = None
        try:
            raw_answer = charge("journal_answer_read", output.read_text)
            chosen = charge(
                "journal_parse",
                lambda: parse_turn_decision(
                    raw_answer,
                    journal_sha256=journal.sha256,
                    canonical_university=case["university"],
                    phase=selector_phase,
                ),
            )
            (case_root / f"selector-{turn:02d}.answer.json").write_text(raw_answer)
            save(case_root / f"selector-{turn:02d}.decision.json", chosen.model_dump(mode="json"))
            if closing_only and chosen.kind == "action":
                raise ValueError("Source navigation is closed; submit observed facts or commit")
            if (
                processed_through < len(actions)
                and chosen.kind == "submit"
                and not chosen.facts
                and not chosen.question_findings
            ):
                raise ValueError(
                    "A pending recorded source requires a data-bearing fact or question finding; hash-only submit cannot process it"
                )
            candidate = charge(
                "journal_candidate",
                lambda: journal.candidate(
                    chosen, actions, ended_at=datetime.now(UTC), case_id=case_id
                ),
            )
            submission = charge("journal_submission", candidate.as_submission)
            expanded = charge(
                "checkpoint_expansion",
                lambda: expand_submission(
                    submission, actions, ended_at=datetime.now(UTC), case_id=case_id
                ),
            )
            proof: list[dict[str, Any]] = []
            if actions and log_dir is not None:
                checkpoint_log_dir = log_dir
                tentative = charge(
                    "checkpoint_trace",
                    lambda: make_trace(
                        case_id,
                        packet,
                        started,
                        actions,
                        expanded["proposals"],
                        expanded["question_findings"],
                        wall_guard=wall_guard,
                    ),
                )
                charge(
                    "checkpoint_raw_log_validation",
                    lambda: validate_trace(tentative, packet, logs_dir=log_dir),
                )
                proof = charge(
                    "checkpoint_all_fact_citations",
                    lambda: validate_all_fact_citations(
                        submission, actions, logs_dir=checkpoint_log_dir, packet=packet
                    ),
                )
            elif submission.facts or submission.question_findings or chosen.kind != "action":
                raise ValueError(
                    "A checkpoint requires an actual recorded controlled source action"
                )
            if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
                result["error"] = "active_budget_exhausted_during_checkpoint_validation"
                break
            event = charge(
                "checkpoint_event_hashing",
                lambda: {
                    "turn": turn,
                    "at": datetime.now(UTC).isoformat(),
                    "decision_sha256": digest(chosen.model_dump(mode="json")),
                    "prior_journal_sha256": journal.sha256,
                    "accepted_journal_sha256": candidate.sha256,
                    "source_action_count": len(actions),
                    "citation_proof_sha256": digest(proof),
                    "selector_mode": selector_mode,
                    "selector_phase": selector_phase,
                    "reasoning_effort": None,
                    "reasoning_effort_source": "cli_default",
                    "selector_schema_sha256": digest(bound_schema),
                    "selector_command_sha256": command_metadata["argv_sha256"],
                    "selector_request_sha256": digest(request_record),
                    "processed_through_action_count": len(actions),
                },
            )
            if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
                result["error"] = "active_budget_exhausted_during_checkpoint_event_hashing"
                break
            candidate_events = [*events, event]
            # One hash-bound marker is written last; only a complete marker commits a delta.
            charge(
                "checkpoint_snapshot_write",
                lambda: save(case_root / f"checkpoint-{turn:02d}.json", candidate.snapshot()),
            )
            if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
                result["error"] = "active_budget_exhausted_during_checkpoint_snapshot_write"
                break
            charge(
                "checkpoint_commit_marker_write",
                lambda: commit_events(case_root / "journal-events.json", candidate_events),
            )
            meter[-1]["turn"] = turn
            if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
                # Retain the earlier marker and an orphan snapshot for audit.
                charge(
                    "checkpoint_failed_commit_restore",
                    lambda: commit_events(case_root / "journal-events.json", events),
                )
                result["error"] = "active_budget_exhausted_during_checkpoint_commit_write"
                break
            journal, events = candidate, candidate_events
            accepted_trace_ended_at = event["at"]
            feedback = None
            emit(
                case=case_id,
                state="checkpoint",
                turn=turn,
                facts=len(submission.facts),
                journal_sha256=journal.sha256,
                active_ms=active_ms,
            )
            if chosen.kind == "submit":
                explicit_commit = True
                result["completion_kind"] = "explicit_commit"
                break
            if chosen.kind == "record":
                continue
            assert chosen.action is not None
            item = charge("action_parse", chosen.action.controller_request)

            def proposal_disposition() -> Any:
                disposition = charge(
                    "navigation_proposal_policy",
                    lambda: source_proposal(wall_guard=wall_guard, source_closed=closing_only),
                )
                meter[-1]["turn"] = turn
                meter[-1]["policy"] = disposition.as_dict()
                return disposition

            def retain_decline(disposition: Any) -> None:
                outcome = charge(
                    "navigation_decline_record_construction",
                    lambda: {
                        "status": "declined",
                        "turn": turn,
                        "decision_sha256": digest(chosen.model_dump(mode="json")),
                        "accepted_journal_sha256": journal.sha256,
                        "source_action_count": len(actions),
                        "processed_through_action_count": events[-1]["source_action_count"],
                        "policy_meter_index": len(meter) - 1,
                        "active_ms": active_ms,
                        **wall_guard.as_dict(),
                        "policy": disposition.as_dict(),
                        "reason": disposition.reason,
                        "cutoff_ms": WALL_GUARD_MS,
                        "action": item,
                    },
                )
                navigation_outcomes.append(outcome)
                meter[-1]["turn"] = turn
                charge(
                    "navigation_outcome_write",
                    lambda: save(case_root / "navigation-outcomes.json", navigation_outcomes),
                )
                meter[-1]["turn"] = turn

            disposition = proposal_disposition()
            if not disposition.dispatch_allowed:
                retain_decline(disposition)
                closing_only = True
                continue
            if item["tool"] == "view":
                step = item["source_step"]
                if log_dir is None or not 1 <= step <= len(actions):
                    raise ValueError("Stored view needs an existing source action")

                def view_saved() -> dict[str, Any]:
                    record = json.loads((log_dir / actions[step - 1]["raw_log_path"]).read_text())
                    return expert_view(
                        record,
                        chars=11000,
                        links=45,
                        sections=60,
                        page=item.get("page"),
                        offset=item.get("offset", 0),
                        section_offset=item.get("section_offset", 0),
                        link_offset=item.get("link_offset", 0),
                    )

                view_basis_ms = active_ms
                view_started_at = datetime.now(UTC).isoformat()
                view = charge("stored_source_view", view_saved)
                view_finished_at = datetime.now(UTC).isoformat()
                view_meter_index = len(meter) - 1
                meter[-1]["turn"] = turn
                view_outcome = charge(
                    "stored_view_receipt_construction",
                    lambda: {
                        "status": "stored_view",
                        "observed_at": disposition.observed_at,
                        "view_started_at": view_started_at,
                        "view_finished_at": view_finished_at,
                        **wall_guard.as_dict(),
                        "turn": turn,
                        "decision_sha256": digest(chosen.model_dump(mode="json")),
                        "accepted_journal_sha256": journal.sha256,
                        "source_action_count": len(actions),
                        "processed_through_action_count": events[-1]["source_action_count"],
                        "active_ms": view_basis_ms,
                        "view_meter_index": view_meter_index,
                        "view_sha256": digest(view),
                        "action": item,
                    },
                )
                navigation_outcomes.append(view_outcome)
                meter[-1]["turn"] = turn
                charge(
                    "navigation_outcome_write",
                    lambda: save(case_root / "navigation-outcomes.json", navigation_outcomes),
                )
                meter[-1]["turn"] = turn
                history.append({"step": step, "pure_saved_view": True, "view": view})
                continue

            def prepare_plan() -> dict[str, Any]:
                plan = {
                    "case_id": case_id,
                    "provider": "exa",
                    "log_prefix": case_id,
                    "strict_budget": True,
                    "official_domains": [case["domain"]],
                    "max_calls": 60,
                    "max_ms": TOTAL_MS,
                    "max_search_queries": 6,
                    "budget_origin": NETWORK_COMPARISON_MODE,
                    "budget_model": BUDGET_MODEL,
                    **wall_guard.as_dict(),
                    "actions": [item],
                }
                if summary:
                    plan["start_calls"] = summary["final"]["fetcher_calls"]
                    plan["start_search_queries"] = summary["final"]["search_queries"]
                    plan["resume_summary"] = {
                        key: value
                        for key, value in summary.items()
                        if key not in {"actions", "replay", "initial"}
                    }
                return plan

            plan = charge("plan_construction", prepare_plan)
            disposition = proposal_disposition()
            if not disposition.dispatch_allowed:
                retain_decline(disposition)
                closing_only = True
                continue
            plan["start_ms"] = source_ms
            inputs = {
                "mode": "expert_actions",
                "search_provider": "exa",
                "plan": json.dumps(plan, ensure_ascii=False),
            }
            if prior_run is not None:
                inputs["resume_run_id"] = str(prior_run)
            round_root = case_root / f"controller-{turn:02d}"
            try:
                try:
                    new_run, actual_sha = charge(
                        "controller_transport",
                        lambda: workflow(
                            repo, ref, case_id, inputs, round_root, wall_guard=wall_guard
                        ),
                    )
                finally:
                    meter[-1]["turn"] = turn
            except TimeoutError as exc:
                if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
                    result["error"] = "wall_guard_exhausted"
                    result["completion_kind"] = "wall_guard_checkpoint"
                    turn_errors.append(
                        {"turn": turn, "kind": "controller_wall_guard", "detail": str(exc)}
                    )
                    break
                STOP.set()
                raise
            except Exception:
                STOP.set()  # Dispatch outcomes are never silently retried.
                raise
            if actual_sha != baseline_result["head_sha"]:
                STOP.set()
                raise RuntimeError("Controller differs from the frozen baseline harness")
            summary_began = time.monotonic()
            try:
                summary = json.loads((round_root / "artifact/summary.json").read_text())
            except Exception:
                elapsed = math.ceil((time.monotonic() - summary_began) * 1000)
                active_ms += elapsed
                meter.append(
                    {
                        "stage": "controller_summary_read_and_parse",
                        "elapsed_ms": elapsed,
                        "active_ms": active_ms,
                    }
                )
                STOP.set()  # An unreadable source clock cannot be safely reset or retried.
                raise
            summary_elapsed = math.ceil((time.monotonic() - summary_began) * 1000)
            if not isinstance(summary, dict):
                STOP.set()
                raise ValueError("Controller summary must be an object")
            prior_run, log_dir = new_run, round_root / "artifact/logs"
            measured = summary["final"]["elapsed_ms"]
            if type(measured) is not int or measured < source_ms:
                STOP.set()
                raise RuntimeError("Controller cumulative clock moved backwards")
            meter.append(
                {
                    "stage": "controlled_source_action",
                    "elapsed_ms": 0,
                    "active_ms": active_ms,
                    "source_elapsed_ms": measured - source_ms,
                    "source_ms": measured,
                    "run_id": new_run,
                    "turn": turn,
                }
            )
            source_ms = measured
            active_ms += summary_elapsed
            meter.append(
                {
                    "stage": "controller_summary_read_and_parse",
                    "elapsed_ms": summary_elapsed,
                    "active_ms": active_ms,
                }
            )
            previous_step = len(actions) or None

            provider_failures = []

            def record_outputs() -> None:
                assert summary is not None and log_dir is not None
                for entry in summary["actions"]:
                    step = len(actions) + 1
                    record = json.loads((log_dir / entry["raw_log_path"]).read_text())
                    if record.get("executed") is True and entry["tool"] == "search_provider":
                        response = record.get("response", {})
                        statuses = {
                            response.get(key)
                            for key in ("http_status", "status_code")
                            if type(response.get(key)) is int
                            and response[key] in {401, 402, 403, 429}
                        }
                        for status in sorted(statuses):
                            provider_failures.append(
                                {
                                    "provider": "exa",
                                    "http_status": status,
                                    "raw_log_path": entry["raw_log_path"],
                                    "raw_log_sha256": entry["raw_log_sha256"],
                                }
                            )
                    action = {
                        key: entry[key]
                        for key in (
                            "at",
                            "tool",
                            "request",
                            "outcome",
                            "outcome_status",
                            "policy_mode",
                            "raw_log_path",
                            "raw_log_sha256",
                            "elapsed_ms",
                            "fetcher_calls_cumulative",
                            "elapsed_ms_cumulative",
                            "search_queries_cumulative",
                        )
                    }
                    action.update(
                        step=step,
                        observed_before=chosen.observation,
                        observed_from_step=previous_step if step > 1 else None,
                        action=f"{item['tool']}: {item.get('url') or item.get('query')}",
                        stage="candidate_generation"
                        if entry["tool"] == "search_provider"
                        else "page_understanding",
                        url=entry["request"].get("url"),
                        final_url=entry.get("final_url"),
                    )
                    actions.append(action)
                    history.append(
                        {
                            "step": step,
                            "view": expert_view(record, chars=11000, links=45, sections=60),
                        }
                    )

            charge("source_output_processing_and_views", record_outputs)
            if provider_failures:
                STOP.set()
                result["provider_failure"] = provider_failures[0]
                result["error"] = f"provider_http_{provider_failures[0]['http_status']}"
                break
            if summary["final"].get("stop_reason") == "http_429":
                STOP.set()
                result["error"] = "provider_http_429"
                break
            emit(
                case=case_id,
                state="observed",
                actions=len(actions),
                active_ms=active_ms,
                reads=summary["final"]["fetcher_calls"],
                queries=summary["final"]["search_queries"],
            )
        except ValueError as exc:

            def rejected_feedback(error: ValueError = exc) -> dict[str, Any]:
                value: dict[str, Any] = {"error": str(error)[:1200]}
                # Diagnostics never commit a delta or bypass a failed journal gate.
                # A structurally invalid delta may also be invalid diagnostic input.
                for key, compute in (
                    (
                        "question_scope_diagnostics",
                        lambda: question_scope_diagnostics(
                            chosen, packet, retained_facts=journal.facts
                        )
                        if chosen is not None
                        else [],
                    ),
                    (
                        "source_location_diagnostics",
                        lambda: location_feedback(chosen, history[-2:])
                        if chosen is not None
                        else None,
                    ),
                ):
                    try:
                        value[key] = compute()
                    except Exception as diagnostic_error:
                        value[key] = [] if key == "question_scope_diagnostics" else None
                        value[key + "_error"] = (
                            type(diagnostic_error).__name__ + ": " + str(diagnostic_error)
                        )[:1200]
                return value

            feedback = charge("rejection_feedback_construction", rejected_feedback)
            turn_errors.append({"turn": turn, "kind": "rejected_checkpoint", "feedback": feedback})
            charge(
                "rejection_feedback_write",
                lambda: save(case_root / f"validation-{turn:02d}.json", {"feedback": feedback}),
            )
            emit(
                case=case_id, state="rejected_checkpoint", turn=turn, detail=feedback["error"][:180]
            )
        except Exception as exc:
            result["error"] = type(exc).__name__ + ": " + str(exc)
            break
    save(case_root / "recorded-actions.json", actions)
    accepted_source_count = events[-1]["source_action_count"] if events else 0
    trace_actions = actions[:accepted_source_count]
    if trace_actions and log_dir is not None:
        try:
            submission = charge("closing_journal_submission", journal.as_submission)
            expanded = charge(
                "closing_expansion",
                lambda: expand_submission(
                    submission, trace_actions, ended_at=datetime.now(UTC), case_id=case_id
                ),
            )
            trace = charge(
                "closing_trace",
                lambda: make_trace(
                    case_id,
                    packet,
                    started,
                    trace_actions,
                    expanded["proposals"],
                    expanded["question_findings"],
                    wall_guard=wall_guard,
                    ended_at=accepted_trace_ended_at,
                ),
            )
            charge(
                "closing_raw_log_validation",
                lambda: validate_trace(trace, packet, logs_dir=log_dir),
            )
            proof = charge(
                "closing_all_fact_citations",
                lambda: validate_all_fact_citations(
                    submission, trace_actions, logs_dir=log_dir, packet=packet
                ),
            )
            charge(
                "closing_journal_write",
                lambda: save(case_root / "accepted-journal.json", journal.snapshot()),
            )
            charge(
                "closing_compact_write",
                lambda: save(
                    case_root / "compact-submission.json", submission.model_dump(mode="json")
                ),
            )
            charge(
                "closing_proof_write",
                lambda: save(case_root / "compact-citation-proof.json", proof),
            )
            charge("closing_trace_write", lambda: save(case_root / "trace.json", trace))
            trace_hash = charge("closing_trace_hash", lambda: digest(trace))
            proof_hash = charge("closing_citation_proof_hash", lambda: digest(proof))
            result.update(
                validated=True,
                trace_sha256=trace_hash,
                trace=str(case_root / "trace.json"),
                journal_sha256=journal.sha256,
                compact_submission=str(case_root / "compact-submission.json"),
                citation_proof_sha256=proof_hash,
                accepted_turns=len(events),
            )
            if events[-1]["source_action_count"] != len(actions):
                result.update(
                    status="incomplete", error=result.get("error") or "unprocessed_recorded_source"
                )
            if not result.get("error") and wall_guard.remaining_seconds(now=datetime.now(UTC)) > 0:
                result["status"] = "submitted"
                result.setdefault("completion_kind", "budget_checkpoint")
            if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0:
                result.update(
                    status="incomplete", error="active_budget_exhausted_during_closing_validation"
                )
        except Exception as exc:
            result.update(validated=False, validation_error=str(exc))
    else:
        result.update(validated=False, error=result.get("error") or "no_recorded_source_action")
    found = charge(
        "terminal_retained_finding_summary",
        lambda: [f.question for f in journal.question_findings if f.status == "found"],
    )
    if wall_guard.remaining_seconds(now=datetime.now(UTC)) <= 0 and result["status"] == "submitted":
        result.update(status="incomplete", error="active_budget_exhausted_during_terminal_summary")
    result.update(
        active_ms=active_ms,
        actions=len(actions),
        logical_expert=f"expert500-{case_id}",
        comparison_mode=NETWORK_COMPARISON_MODE,
        budget_model=BUDGET_MODEL,
        **wall_guard.as_dict(),
        provider="exa",
        requested_model=MODEL,
        requested_reasoning_effort=None,
        reasoning_effort_source="cli_default",
        turn_policy=policy_limits(),
        network_budget=expert_budget(),
        navigation_outcomes=navigation_outcomes,
        exact_metering=True,
        codex_cli=CLI,
        source_ms=source_ms,
        wall_elapsed_ms=math.ceil((datetime.now(UTC) - started_dt).total_seconds() * 1000),
        accepted_trace_ended_at=accepted_trace_ended_at,
        accepted_source_action_count=accepted_source_count,
        recorded_actions_path=str(case_root / "recorded-actions.json"),
        recorded_actions_sha256=hashlib.sha256(
            (case_root / "recorded-actions.json").read_bytes()
        ).hexdigest(),
        finished_at=datetime.now(UTC).isoformat(),
        turn_errors=turn_errors,
        explicit_commit=explicit_commit,
        protocol_version="journal-v8",
        processed_through_action_count=events[-1]["source_action_count"] if events else 0,
        all_recorded_sources_processed=bool(events)
        and events[-1]["source_action_count"] == len(actions),
        real_elapsed_seconds=(datetime.now(UTC) - datetime.fromisoformat(started)).total_seconds(),
        active_time_basis="descriptive wall-work meter only; never charged into source ledger",
        network_time_basis="measured controller actions only; excludes model, mechanics and transport",
        meter=meter,
        found=found,
    )
    save(case_root / "result.json", result)
    emit(
        case=case_id,
        state=result["status"],
        completion_kind=result.get("completion_kind"),
        validated=result["validated"],
        found=result["found"],
        actions=len(actions),
        active_ms=active_ms,
    )
    return result


def main() -> None:
    parser = argparse.ArgumentParser()
    parser.add_argument("--repo", type=Path, required=True)
    parser.add_argument("--root", type=Path, required=True)
    parser.add_argument("--ref", required=True)
    parser.add_argument("--stage", choices=["baseline", "experts"], required=True)
    args = parser.parse_args()
    trial = json.loads((args.root / "trial.json").read_text())
    if (
        trial["protocol_version"] != "journal-v8"
        or trial["frozen_ref"] != args.ref
        or trial.get("algorithm_provider") != "exa"
        or trial.get("expert_provider") != "exa"
        or trial.get("search_provider") != "exa"
    ):
        raise ValueError("Stage does not match original frozen V8 trial")
    for name, field in (
        ("workspace-locations.json", "workspace_assignments_sha256"),
        ("workspace-receipts.json", "workspace_receipts_sha256"),
    ):
        if hashlib.sha256((args.root / name).read_bytes()).hexdigest() != trial[field]:
            raise ValueError("Original independently allocated workspace contract changed")
    assignments = json.loads((args.root / "workspace-locations.json").read_text())
    receipts = json.loads((args.root / "workspace-receipts.json").read_text())
    workspaces = validate_workspaces(assignments, receipts, case_ids=tuple(legacy["COHORT"]))
    if args.stage == "experts":
        baselines = json.loads((args.root / "baseline-results.json").read_text())
        if (
            baselines["stopped"]
            or baselines["remaining_case_ids"]
            or any(
                result.get("provider") != "exa"
                or result.get("provider_failure") is not None
                or result.get("budget", {}).get("provider_failure") is not None
                for result in baselines["results"]
            )
        ):
            raise ValueError("Provider-stopped/incomplete baseline cannot start experts")
    if (args.root / f"{args.stage}-results.json").exists():
        raise ValueError("Do not reset or retry a stage in an existing trial")
    (args.root / "sealed-sentinel.txt").write_text("SEALED_HARMLESS_SENTINEL")
    results = []
    cohort = legacy["COHORT"]
    with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
        pending = {}
        iterator = iter(cohort)

        def submit_next() -> None:
            if STOP.is_set():
                return
            case_id = next(iterator, None)
            if case_id is None:
                return
            function: Callable[..., dict[str, Any]] = (
                baseline if args.stage == "baseline" else expert
            )
            params = (args.repo, args.ref, args.root, case_id)
            future = pool.submit(
                function, *params, *(() if args.stage == "baseline" else (workspaces,))
            )
            pending[future] = case_id

        for _ in range(3):
            submit_next()
        while pending:
            done, _ = concurrent.futures.wait(
                pending, return_when=concurrent.futures.FIRST_COMPLETED
            )
            for future in done:
                case_id = pending.pop(future)
                try:
                    results.append(future.result())
                except Exception as exc:
                    failure = {
                        "case_id": case_id,
                        "status": "failed",
                        "error": type(exc).__name__ + ": " + str(exc),
                    }
                    results.append(failure)
                    save(args.root / args.stage / case_id / "orchestration-error.json", failure)
                    emit(**failure)
                    STOP.set()
            if not STOP.is_set():
                for _ in done:
                    submit_next()
            save(
                args.root / f"{args.stage}-results.json",
                {
                    "results": results,
                    "stopped": STOP.is_set(),
                    "remaining_case_ids": [
                        case_id
                        for case_id in cohort
                        if case_id not in {r["case_id"] for r in results}
                    ],
                },
            )
    emit(stage=args.stage, completed=len(results), stopped=STOP.is_set())


if __name__ == "__main__":
    main()
