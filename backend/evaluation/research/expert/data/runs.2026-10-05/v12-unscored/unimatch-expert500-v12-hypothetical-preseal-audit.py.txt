observations = []

def hypothetical_policy(folder, result):
    copied = dict(result)
    expected = math.ceil((datetime.fromisoformat(result['finished_at']) -
                          datetime.fromisoformat(result['wall_started_at'])).total_seconds() * 1000)
    observations.append({'case': result['case_id'], 'original_wall_elapsed_ms': result['wall_elapsed_ms'],
                         'hypothetical_timestamp_derived_ms': expected,
                         'original_result_not_changed': True})
    copied['wall_elapsed_ms'] = expected
    return original(folder, copied)

class StopBeforeSeal(Exception):
    pass

def no_seal(path, seal):
    assert not path.exists()
    assert seal['count'] == 10
    raise StopBeforeSeal()

ns['verify_turn_policy_evidence'] = hypothetical_policy
ns['immutable_seal'] = no_seal
sys.argv = [str(REPORT), '--root', str(ROOT), '--dataset',

            '--orchestrator', '/tmp/unimatch-expert500-journal-pilot-v12.py']
try:
    ns['main']()
except StopBeforeSeal:
