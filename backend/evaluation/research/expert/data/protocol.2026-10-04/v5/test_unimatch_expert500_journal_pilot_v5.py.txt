def source_action(tool="fetcher", **changes):
    value = {
        "tool": tool,
        "url": URL if tool == "fetcher" else None,
        "query": "site:u.edu official admissions" if tool == "search_provider" else None,
        "domains": ["u.edu"] if tool == "search_provider" else [],
        "max_results": 4 if tool == "search_provider" else None,
        "source_step": 1 if tool == "view" else None,
        "page": None,
        "offset": None,
        "section_offset": None,
        "link_offset": None,
    }
    return {**value, **changes}


def fact(**changes):
    value = {
        "id": "english",
        "question": "Q1",
        "claim_key": "ielts.overall",
        "value_json": "6.5",
        "source_step": 1,
        "quotation": QUOTE,
        "location": {"kind": "text", "block_id": None, "section_path": [], "page": None},
        "scope": dict(SCOPE),
    }
    return {**value, **changes}


def finding(**changes):
    value = {
        "question": "Q1",
        "status": "found",
        "fact_ids": ["english"],
        "scope": dict(SCOPE),
        "observation": "The recorded synthetic page states this minimum.",
        "supporting_steps": [],
    }
    return {**value, **changes}


def decision(prompt, kind="action", facts=None, findings=None, action=None, **changes):
    if "accepted_journal" not in prompt:
        assert kind == "submit" and action is None and not facts and not findings
        return minimal_commit(prompt, **changes)
    value = {
        "kind": kind,
        "observation": "Inspect the current recorded official source.",
        "journal_sha256": prompt["accepted_journal"]["sha256"],
        "action": source_action() if kind == "action" and action is None else action,
        "facts": facts or [],
        "question_findings": findings or [],
    }
    return {**value, **changes}


def minimal_commit(prompt, **changes):
    from evaluation.research.expert.representation import CLOSING_OBSERVATION

    current_hash = (
        prompt["accepted_journal"]["sha256"]
        if "accepted_journal" in prompt
        else prompt["journal_sha256"]
    )
    return {
        "kind": "submit",
        "observation": CLOSING_OBSERVATION,
        "journal_sha256": current_hash,
        "action": None,
        "facts": [],
        "question_findings": [],
        **changes,
    }


def completed_events(answer):
    return (
        "\n".join(
            json.dumps(item)
            for item in [
                {"type": "thread.started", "thread_id": "synthetic"},
                {"type": "turn.started"},
                {
                    "type": "item.completed",
                    "item": {"id": "answer", "type": "agent_message", "text": answer},
                },
                {"type": "turn.completed", "usage": {"input_tokens": 1, "output_tokens": 1}},
            ]
        )
        + "\n"
    )


@dataclass
class Harness:
    runtime: dict
    repo: Path
    root: Path
    workspace: Path
    packet: dict
    pending: list = field(default_factory=list)
    calls: list = field(default_factory=list)
    queries: list = field(default_factory=list)
    prompts: list = field(default_factory=list)
    commands: list = field(default_factory=list)
    plans: list = field(default_factory=list)
    controller_dirs: list = field(default_factory=list)
    summary_hook: object = None
    workflow_error: Exception | None = None
    source_body: bytes | None = None
    source_content_type: str = "text/html"
    plan_hook: object = None

    @property
    def folder(self):
        return self.root / "experts/u"

    def queue(self, callback=None, *, elapsed=12000, error=None, raw=None):
        self.pending.append({"callback": callback, "elapsed": elapsed, "error": error, "raw": raw})

    def read(self, name):
        return json.loads((self.folder / name).read_text())

    def run(self):
        return self.runtime["expert"](
            self.repo, "synthetic-frozen", self.root, "u", {"u": self.workspace}
        )

def bootstrap(pilot):
    pilot.queue()


def checkpoint(pilot, *, kind="record", elapsed=12000, **changes):
    pilot.queue(
        lambda prompt: decision(prompt, kind, facts=[fact()], findings=[finding()], **changes),
        elapsed=elapsed,
    )


def commit(pilot, *, elapsed=500, acknowledge_source=None):
    pilot.queue(
        lambda prompt: decision(
            prompt,
            "submit",
            facts=[
                fact(id="english_second_source", question="OTHER", source_step=acknowledge_source)
            ]
            if acknowledge_source is not None
            else [],
        ),
        elapsed=elapsed,
    )


def assert_exact_clock(result):
    cumulative = 0
    for entry in result["meter"]:
        assert type(entry["elapsed_ms"]) is int and entry["elapsed_ms"] >= 0
        cumulative += entry["elapsed_ms"]
        assert entry["active_ms"] == cumulative
    assert result["active_ms"] == cumulative

@pytest.mark.parametrize(
    "slow_stage", ["controller_summary_parse", "checkpoint_hash", "final_proof_hash"]
)
def test_slow_source_processing_and_hashes_are_charged_before_completion(pilot, slow_stage):
    bootstrap(pilot)
    checkpoint(pilot)
    commit(pilot)
    clock = SimpleNamespace(value=0.0)
    pilot.runtime["time"] = SimpleNamespace(monotonic=lambda: clock.value)
    original_digest = pilot.runtime["digest"]
    injected = False
    proof_hashes = 0

    def slow_digest(value):
        nonlocal injected, proof_hashes
        if (
            isinstance(value, list)
            and value
            and isinstance(value[0], dict)
            and "fact_id" in value[0]
        ):
            proof_hashes += 1
        target = (
            slow_stage == "checkpoint_hash"
            and isinstance(value, dict)
            and value.get("kind") == "record"
        )
        target = target or (
            slow_stage == "final_proof_hash"
            and proof_hashes == 3
            and isinstance(value, list)
            and value
            and "fact_id" in value[0]
        )
        if target and not injected:
            injected = True
            clock.value += 110.0
        return original_digest(value)

    def slow_loads(value):
        nonlocal injected
        parsed = json.loads(value)
        if (
            slow_stage == "controller_summary_parse"
            and not injected
            and isinstance(parsed, dict)
            and {"actions", "tool_policy", "final", "raw_logs"} <= set(parsed)
        ):
            injected = True
            clock.value += 110.0
        return parsed

    pilot.runtime["digest"] = slow_digest
    pilot.runtime["json"] = SimpleNamespace(loads=slow_loads, dumps=json.dumps)
    result = pilot.run()
    assert injected
    assert result["active_ms"] >= 120000 and result["status"] == "incomplete"
    assert pilot.calls == [URL]
    if slow_stage == "checkpoint_hash":
        assert pilot.read("accepted-journal.json")["facts"] == []
    assert_exact_clock(result)

def close_after_source(pilot, source_count=1):
    def exhausted(summary, plan):
        if len(pilot.plans) == source_count:

    for turn, phase, cap, reserve in (
        (1, "bootstrap", 30000, 13000),
        (2, "data", 55000, 13000),

    ):
        request = pilot.read(f"selector-{turn:02d}.request.json")
        meta = pilot.read(f"selector-{turn:02d}.meta.json")

        expected = min(cap, 120000 - start - reserve)

        assert meta["selector_timeout_ms"] == expected
        assert (
            meta["selector_timeout_seconds"]
            == pilot.prompts[turn - 1][1]["timeout_seconds"]
            == expected / 1000
        )
        assert start + expected + reserve <= 120000
        assert request["selector_command"] == pilot.commands[turn - 1]
        argv = request["selector_command"]
        assert argv[argv.index("--model") + 1] == "gpt-6-astra"
        assert argv.count('model_reasoning_effort="low"') == 1
        assert (
            request["command_proof"]["model_reasoning_effort"] == meta["reasoning_effort"] == "low"
        )
        assert request["schema_sha256"] == meta["schema_sha256"] == digest(request["schema"])
        assert events[turn - 1]["selector_phase"] == phase
        assert (
            events[turn - 1]["selector_command_sha256"] == request["command_proof"]["argv_sha256"]
        )
        assert request["schema"]["properties"]["facts"]["maxItems"] == (0 if turn == 3 else 4)
        assert request["schema"]["properties"]["question_findings"]["maxItems"] == (
            0 if turn == 3 else 2

def test_same_source_location_feedback_keeps_rejected_delta_out_until_corrected(pilot):
    bootstrap(pilot)
    checkpoint(pilot)
    invalid = fact(
        id="located",
        question="OTHER",
        claim_key="public.synthetic.note",
        value_json=json.dumps("IELTS policy"),
        location={"kind": "document_ir", "block_id": "b99999", "section_path": [], "page": None},
    )
    pilot.queue(lambda prompt: decision(prompt, "record", facts=[invalid]))
    observed = {}

    def corrected(prompt):
        feedback = prompt["validation_feedback"]["source_location_diagnostics"]
        observed.update(copy.deepcopy(feedback))
        diagnostic = next(item for item in feedback["diagnostics"] if item["fact_id"] == "located")
        candidate = next(
            item for item in diagnostic["candidates"] if item["quotation_in_block_text"]
        )
        assert diagnostic["source_step"] == 1 and candidate["match_kind"] == "exact_block_text"
        assert QUOTE in candidate["text_window"]
        fixed = {**invalid, "location": candidate["location"]}
        return decision(prompt, "record", facts=[fixed])

    pilot.queue(corrected)
    commit(pilot)
    result = pilot.run()
    assert result["validated"] and result["turn_errors"][0]["kind"] == "rejected_checkpoint"
    assert [event["turn"] for event in pilot.read("journal-events.json")] == [1, 2, 4, 5]
    assert not (pilot.folder / "checkpoint-03.json").exists()
    assert pilot.read("checkpoint-02.json")["facts"] == [fact()]
    assert [item["id"] for item in pilot.read("accepted-journal.json")["facts"]] == [
        "english",
        "located",
    ]
    assert observed["offending_fact_ids"] == ["located"]
    assert pilot.calls == [URL] and pilot.queries == []
    assert any(item["stage"] == "rejection_feedback_construction" for item in result["meter"])
    assert any(item["stage"] == "rejection_feedback_write" for item in result["meter"])
