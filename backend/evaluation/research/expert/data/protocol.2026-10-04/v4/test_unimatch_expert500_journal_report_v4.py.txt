def write(path, value):
    path.write_text(json.dumps(value))


def prepare_reporting(
    pilot, report, *, failed=False, rejected=False, minimal=False, next_source=False
):
    if minimal:
        close_after_source(pilot, 2 if next_source else 1)
    bootstrap(pilot)
    checkpoint(pilot, kind="action" if next_source else "record")
    if rejected:
        pilot.queue(
            lambda prompt: decision(
                prompt,
                "record",
                facts=[fact(id="bad", question="OTHER", quotation="Absent synthetic quotation")],
            )
        )
    if failed:
        pilot.queue(error="selector_timeout", elapsed=22013)
    else:
        commit(pilot, acknowledge_source=2 if next_source else None)
    result = pilot.run()
    assert result["validated"]
    write(pilot.root / "experts-results.json", {"stopped": False, "results": [result]})
    baseline_dir = pilot.root / "baseline/u"
    budget = {
        "fetcher_calls": 0,
        "search_queries": 0,
        "elapsed_ms": 1,
        "stop_reason": None,
        "finalization_reserve_ms": 1000,
        "source_work_deadline_ms": 119000,
        "max_active_seconds": 120,
        "max_fetcher_calls": 60,
        "max_search_queries": 6,
        "process_error": None,
        "active_checkpoint_available": True,
    }
    capture = Capture(
        pipeline_sha=HEAD,
        captured_at="2026-10-04T00:00:00+00:00",
        mode="live",  # Schema-compatible mocked bytes; no live calls are made.
        config={
            "synthetic_fixture": True,
            "finalization_reserve_ms": 1000,
            "source_work_deadline_ms": 119000,
            "seconds_per_case": 120,
            "child_active_seconds": 120,
            "strict_expert_budget": True,
            "search_provider": "exa_mcp",
            "case_budgets": {"u": dict(budget)},
        },
        observations=[Observation(case_id="u")],
    )
    capture_path = baseline_dir / "capture.json"
    write(capture_path, capture.model_dump(mode="json"))
    baseline = json.loads((baseline_dir / "result.json").read_text())
    baseline.update(
        capture=str(capture_path),
        run_id=777,
        budget=budget,
    )
    write(baseline_dir / "result.json", baseline)
    scripts = {
        str(path): report["file_hash"](path)
        for path in (PILOT, REPORT, Path("/tmp/unimatch-expert500-pilot-v1-frozen.py"))
    }
    modules = {
        name: report["file_hash"](Path(importlib.util.find_spec(name).origin)) for name in MODULES
    }
    write(
        pilot.root / "trial.json",
        {
            "protocol_version": "journal-v4",
            "baseline_finalization_reserve_ms": 1000,
            "binding_registry": {
                "path": str(report["synthetic_binding_path"].resolve()),
                "sha256": report["file_hash"](report["synthetic_binding_path"]),
            },
            "harness_sha": HEAD,
            "scripts": scripts,
            "modules": modules,
        },
    )
    (pilot.root / "trial-at-baseline-start.json").write_bytes(
        (pilot.root / "trial.json").read_bytes()
    )
    return result

def run_without_labels(pilot, report, monkeypatch):
    """Exercise the real report main while its scorer sees no labels at all."""
    dataset = pilot.root / "no-labels-sentinel.txt"
    dataset.write_text("SYNTHETIC_NO_REVIEWED_LABELS")
    reads = []

    class NoLabels:
        @classmethod
        def model_validate_json(cls, value):
            if isinstance(value, bytes):
                value = value.decode()
            assert value == "SYNTHETIC_NO_REVIEWED_LABELS"
            assert (pilot.root / "submission-seal.json").is_file()
            reads.append(value)
            return None

    def no_label_score(dataset, capture):
        assert dataset is None
        return {
            "metrics": {
                "claim_recall": {"numerator": 0, "denominator": 0, "value": None},
                "claim_precision": {"numerator": 0, "denominator": 0, "value": None},
            },
            "cases": [{"case_id": observation.case_id} for observation in capture.observations],
        }

    report.update(Dataset=NoLabels, score=no_label_score)
    monkeypatch.setattr(
        sys,
        "argv",
        [
            "journal-report",
            "--root",
            str(pilot.root),
            "--dataset",
            str(dataset),
            "--orchestrator",
            str(PILOT),
        ],
    )
    return reads

@pytest.mark.parametrize(
    "change",
    ["tool", "failed", "unfinished", "second_completion", "different_answer", "duplicate_json"],
)
def test_original_selector_provenance_is_required_for_checkpoint_replay(pilot, report, change):
    prepare_reporting(pilot, report)
    path = pilot.folder / "selector-02.events.jsonl"
    raw = (pilot.folder / "selector-02.answer.json").read_text()
    rows = [json.loads(line) for line in completed_events(raw).splitlines()]
    if change == "tool":
        rows.insert(
            2,
            {
                "type": "item.completed",
                "item": {"id": "tool", "type": "command_execution", "command": "cat sealed"},
            },
        )
    elif change == "failed":
        rows[-1] = {"type": "turn.failed", "error": {"message": "Synthetic failure"}}
    elif change == "unfinished":
        rows.pop()
    elif change == "second_completion":
        rows.append(copy.deepcopy(rows[-1]))
    elif change == "different_answer":
        changed = json.loads(raw)
        changed["observation"] = "A different completed decision."
        rows[2]["item"]["text"] = json.dumps(changed)
    else:
        duplicate = raw.replace('"kind": "record"', '"kind": "record", "kind": "submit"', 1)
        assert duplicate != raw
        (pilot.folder / "selector-02.answer.json").write_text(duplicate)
        rows[2]["item"]["text"] = duplicate
    path.write_text("\n".join(json.dumps(row) for row in rows) + "\n")
    with pytest.raises(ValueError):
        replay(pilot, report)

@pytest.mark.parametrize(
    "change", ["missing_exact", "empty", "negative", "float", "bool", "prefix", "final"]
)
def test_full_cumulative_meter_rejects_non_exact_ledgers(report, change):
    result = {
        "exact_metering": True,
        "active_ms": 3,
        "meter": [{"elapsed_ms": 1, "active_ms": 1}, {"elapsed_ms": 2, "active_ms": 3}],
    }
    assert report["exact_metering"](result)
    if change == "missing_exact":
        result.pop("exact_metering")
    elif change == "empty":
        result["meter"] = []
    elif change == "negative":
        result["meter"][0]["elapsed_ms"] = -1
    elif change == "float":
        result["meter"][0]["elapsed_ms"] = 1.0
    elif change == "bool":
        result["meter"][0]["elapsed_ms"] = True
    elif change == "prefix":
        result["meter"][0]["active_ms"] = 2
    else:
        result["active_ms"] = 4
    assert not report["exact_metering"](result)

def test_seal_is_immutable_except_for_ignored_new_sealed_at(report, tmp_path):
    path = tmp_path / "seal.json"
    before = {"sealed_at": "old", "count": 1, "submissions": [{"sha": "a"}]}
    report["immutable_seal"](path, before)
    old_bytes = path.read_bytes()
    assert report["immutable_seal"](path, {**before, "sealed_at": "new"}) == before
    with pytest.raises(ValueError):
        report["immutable_seal"](path, {**before, "submissions": [{"sha": "b"}]})
    assert path.read_bytes() == old_bytes

def update_baseline(pilot, *, budget_changes=None, config_changes=None, observation_error=None):
    folder = pilot.root / "baseline/u"
    result_path, capture_path = folder / "result.json", folder / "capture.json"
    baseline = json.loads(result_path.read_text())
    capture = json.loads(capture_path.read_text())
    for key, value in (budget_changes or {}).items():
        if value is None:
            baseline["budget"].pop(key, None)
        else:
            baseline["budget"][key] = value
    for key, value in (config_changes or {}).items():
        if value is None:
            capture["config"].pop(key, None)
        else:
            capture["config"][key] = value
    capture["config"]["case_budgets"]["u"] = copy.deepcopy(baseline["budget"])
    if observation_error is not None:
        capture["observations"][0]["error"] = observation_error
    write(result_path, baseline)
    write(capture_path, capture)
    return baseline["budget"], Capture.model_validate(capture)

@pytest.mark.parametrize("change", ["outer_bytes", "payload_hash", "source_quote"])
def test_raw_source_tampering_is_rejected_before_any_seal_or_label_read(
    pilot, report, monkeypatch, change
):
    prepare_reporting(pilot, report, minimal=True)
    trace = pilot.read("trace.json")
    latest = pilot.controller_dirs[-1] / "artifact"
    relative = trace["actions"][0]["raw_log_path"]
    path = latest / "logs" / relative
    if change == "outer_bytes":
        path.write_bytes(path.read_bytes() + b"\n")
    else:
        record = json.loads(path.read_text())
        if change == "payload_hash":
            record["response"]["content_sha256"] = "0" * 64
        else:
            body = b"<p>International bachelor applicants need IELTS Academic overall 7.0.</p>"
            record["response"]["content_b64"] = base64.b64encode(body).decode()
            record["response"]["content_sha256"] = hashlib.sha256(body).hexdigest()
        write(path, record)
        new_hash = report["file_hash"](path)
        trace["actions"][0]["raw_log_sha256"] = new_hash
        result = pilot.read("result.json")
        result["trace_sha256"] = report["digest"](trace)
        summary = json.loads((latest / "summary.json").read_text())
        summary["actions"][0]["raw_log_sha256"] = new_hash
        summary["raw_logs"][relative] = new_hash
        write(latest / "summary.json", summary)
        write(pilot.folder / "trace.json", trace)
        write(pilot.folder / "result.json", result)
        write(pilot.root / "experts-results.json", {"stopped": False, "results": [result]})
    reads = run_without_labels(pilot, report, monkeypatch)
    with pytest.raises(ValueError):
        report["main"]()
    assert reads == [] and not (pilot.root / "submission-seal.json").exists()


def test_identity_content_and_label_interface_are_loaded_only_after_the_immutable_seal(
    pilot, report, monkeypatch
):
    prepare_reporting(pilot, report)
    reads = run_without_labels(pilot, report, monkeypatch)
    original_identities, original_dataset = report["IdentityMap"], report["Dataset"]
    order = []
    original_read_bytes = Path.read_bytes
    post_seal_binding_reads = []

    def observed_read_bytes(path):
        if (
            path == report["synthetic_binding_path"]
            and (pilot.root / "submission-seal.json").is_file()
        ):
            post_seal_binding_reads.append(path)
        return original_read_bytes(path)

    monkeypatch.setattr(Path, "read_bytes", observed_read_bytes)

    class SealedIdentities:
        @classmethod
        def model_validate_json(cls, value):
            assert (pilot.root / "submission-seal.json").is_file()
            order.append("identities")
            return original_identities.model_validate_json(value)

    class SealedLabels:
        @classmethod
        def model_validate_json(cls, value):
            assert order == ["identities"]
            order.append("labels")
            return original_dataset.model_validate_json(value)

    report.update(IdentityMap=SealedIdentities, Dataset=SealedLabels)
    report["main"]()
    assert order == ["identities", "labels"] and reads == ["SYNTHETIC_NO_REVIEWED_LABELS"]
    assert post_seal_binding_reads == [report["synthetic_binding_path"]]
