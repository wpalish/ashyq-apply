def close_after_source(pilot, source_count=1):
    """Reach the genuine charged cutoff; preserve the controller summary/raw bytes."""
    clock = SimpleNamespace(value=0.0)
    pilot.runtime["time"] = SimpleNamespace(monotonic=lambda: clock.value)
    captured = {"elapsed_ms": 0, "done": False}
    original = pilot.runtime["expert_view"]

    def observed(summary, plan):
        captured["elapsed_ms"] = summary["final"]["elapsed_ms"]

    def view(record, **kwargs):
        if len(pilot.plans) == source_count and not captured["done"]:
            clock.value += max(0, 49000 - captured["elapsed_ms"]) / 1000
            captured["done"] = True
        return original(record, **kwargs)

    pilot.summary_hook = observed
    pilot.runtime["expert_view"] = view


def prepare_reporting(
    pilot,
    report,
    *,
    failed=False,
    rejected=False,
    minimal=False,
    next_source=False,
    stored_view=False,
    declined=False,

    if minimal:
        close_after_source(pilot, 2 if next_source else 1)
    bootstrap(pilot)
    if stored_view:

                "tool": "view",

    baseline_dir = pilot.root / "baseline/u"

def run_without_labels(pilot, report, monkeypatch):
    """Exercise the real report main while its scorer sees no labels at all."""
    dataset = pilot.root / "no-labels-sentinel.txt"
    dataset.write_text("SYNTHETIC_NO_REVIEWED_LABELS")
    reads = []

    class NoLabels:
        @classmethod
        def model_validate_json(cls, value):
            if isinstance(value, bytes):
                value = value.decode()
            assert value == "SYNTHETIC_NO_REVIEWED_LABELS"
            assert (pilot.root / "submission-seal.json").is_file()
            reads.append(value)
            return None

    def no_label_score(dataset, capture):
        assert dataset is None
        return {
            "metrics": {
                "claim_recall": {"numerator": 0, "denominator": 0, "value": None},
                "claim_precision": {"numerator": 0, "denominator": 0, "value": None},
            },
            "cases": [{"case_id": observation.case_id} for observation in capture.observations],
        }

    report.update(Dataset=NoLabels, score=no_label_score)
    monkeypatch.setattr(
        sys,
        "argv",
        [
            "journal-report",
            "--root",
            str(pilot.root),
            "--dataset",
            str(dataset),
            "--orchestrator",
            str(PILOT),
        ],
    )
    return reads

def update_baseline(pilot, *, budget_changes=None, config_changes=None, observation_error=None):
    folder = pilot.root / "baseline/u"
    result_path, capture_path = folder / "result.json", folder / "capture.json"
    baseline = json.loads(result_path.read_text())
    capture = json.loads(capture_path.read_text())
    for key, value in (budget_changes or {}).items():
        if value is None:
            baseline["budget"].pop(key, None)
        else:
            baseline["budget"][key] = value
    for key, value in (config_changes or {}).items():
        if value is None:
            capture["config"].pop(key, None)
        else:
            capture["config"][key] = value
    capture["config"]["case_budgets"]["u"] = copy.deepcopy(baseline["budget"])
    if observation_error is not None:
        capture["observations"][0]["error"] = observation_error
    write(result_path, baseline)
    write(capture_path, capture)
    return baseline["budget"], Capture.model_validate(capture)

def test_identity_content_and_label_interface_are_loaded_only_after_the_immutable_seal(
    pilot, report, monkeypatch
):
    prepare_reporting(pilot, report)
    reads = run_without_labels(pilot, report, monkeypatch)
    original_identities, original_dataset = report["IdentityMap"], report["Dataset"]
    order = []
    original_read_bytes = Path.read_bytes
    post_seal_binding_reads = []

    def observed_read_bytes(path):
        if (
            path == report["synthetic_binding_path"]
            and (pilot.root / "submission-seal.json").is_file()
        ):
            post_seal_binding_reads.append(path)
        return original_read_bytes(path)

    monkeypatch.setattr(Path, "read_bytes", observed_read_bytes)

    class SealedIdentities:
        @classmethod
        def model_validate_json(cls, value):
            assert (pilot.root / "submission-seal.json").is_file()
            order.append("identities")
            return original_identities.model_validate_json(value)

    class SealedLabels:
        @classmethod
        def model_validate_json(cls, value):
            assert order == ["identities"]
            order.append("labels")
            return original_dataset.model_validate_json(value)

    report.update(IdentityMap=SealedIdentities, Dataset=SealedLabels)
    report["main"]()
    assert order == ["identities", "labels"] and reads == ["SYNTHETIC_NO_REVIEWED_LABELS"]
    assert post_seal_binding_reads == [report["synthetic_binding_path"]]

@pytest.mark.parametrize("after", ["next_state", "closing_selector"])
def test_future_decline_cannot_close_an_earlier_research_state_or_minimal_turn(
    pilot, report, after
):
    result = prepare_reporting(pilot, report, declined=True)
    assert replay(pilot, report)["explicit_commit"]
    entries = result["meter"]
    outcomes = pilot.read("navigation-outcomes.json")
    outcome = outcomes[0]
    policy_index = outcome["policy_meter_index"]
    delayed = entries[policy_index : policy_index + 3]
    assert [entry["stage"] for entry in delayed] == [
        "navigation_proposal_policy",
        "navigation_decline_record_construction",
        "navigation_outcome_write",
    ]
    del entries[policy_index : policy_index + 3]
    following = next(
        i
        for i, entry in enumerate(entries)
        if entry["stage"]
        == ("research_state_policy" if after == "next_state" else "closing_selector")
        and entry.get("turn") == 3
    )
    insert_at = following + 1
    entries[insert_at:insert_at] = delayed
    outcome["policy_meter_index"] = insert_at
    # The early20s minimal turn appears closed only if replay trusts a future
    # decline's turn label without its construction/write chronology.
    state_meter = next(
        entry
        for entry in entries
        if entry["stage"] == "research_state_policy" and entry.get("turn") == 3
    )
    state_meter.update(elapsed_ms=0, active_ms=20000)
    selector_index = next(
        i
        for i, entry in enumerate(entries)
        if entry["stage"] == "closing_selector" and entry.get("turn") == 3
    )
    written = [
        entry for entry in entries[:selector_index] if entry["stage"] == "selector_request_write"
    ][-1]
    written.update(elapsed_ms=0, active_ms=20000)
    entries[selector_index]["active_ms"] = 20000 + entries[selector_index]["elapsed_ms"]
    request = pilot.read("selector-03.request.json")
    request["research_state"] = report["research_state"](
        active_ms=20000,
        recorded_source_count=1,
        processed_source_count=1,
        source_closed=True,
    ).as_dict()
    request["selector_policy_active_ms"] = 20000
    meta = pilot.read("selector-03.meta.json")
    meta["selector_start_active_ms"] = 20000
    window = report["turn_window"](phase="minimal_commit", active_ms=20000)
    meta.update(
        selector_timeout_ms=window.timeout_ms, selector_timeout_seconds=window.timeout_seconds
    )
    events = pilot.read("journal-events.json")
    events[-1]["selector_request_sha256"] = report["digest"](request)
    write(pilot.folder / "selector-03.request.json", request)
    write(pilot.folder / "selector-03.meta.json", meta)
    write(pilot.folder / "journal-events.json", events)
    write(pilot.folder / "navigation-outcomes.json", outcomes)
    result["navigation_outcomes"] = outcomes
    with pytest.raises(ValueError) as rejected:
        report["verify_turn_policy_evidence"](pilot.folder, result)
        report["verify_navigation_outcomes"](
            pilot.folder,
            pilot.read("trace.json"),
            pilot.controller_dirs[-1] / "artifact/logs",
            events,
            result,
        )
    assert any(
        word in str(rejected.value).lower()
        for word in ("chronology", "cutoff", "navigation", "decline", "research routing")
    ), str(rejected.value)

@pytest.mark.parametrize(
    "change",
    [
        "raw_deadline_95000",
        "all_deadlines_95000",
        "deadline_float",
        "provider_exa",
        "foreign_phase",
        "foreign_case",
        "foreign_prefix",
        "foreign_origin",
        "strict_budget_integer",
        "max_calls_61",
        "max_ms_120001",
        "max_queries_7",
        "max_queries_float",
        "raw_policy_digest_mismatch",
    ],
)
def test_original_raw_controller_policy_is_bound_to_dispatch_despite_rehashed_metadata(
    pilot, report, monkeypatch, change
):
    prepare_reporting(pilot, report)
    directory = pilot.controller_dirs[-1]
    assert report["verified_controller_stop"](directory) is None
    summary_path = directory / "artifact/summary.json"
    summary = json.loads(summary_path.read_text())
    dispatch_path = directory / "dispatch.json"
    dispatch = json.loads(dispatch_path.read_text())
    plan = json.loads(dispatch["inputs"]["plan"])
    relative = summary["actions"][0]["raw_log_path"]
    raw_path = directory / "artifact/logs" / relative
    raw = json.loads(raw_path.read_text())
    original_response = copy.deepcopy(raw["response"])
    if change in {"raw_deadline_95000", "all_deadlines_95000", "deadline_float"}:
        raw["tool_policy"]["action_deadline_ms"] = 49000.0 if change == "deadline_float" else 95000
        if change == "all_deadlines_95000":
            summary["action_deadline_ms"] = plan["action_deadline_ms"] = 95000
    elif change == "provider_exa":
        raw["provider"] = summary["provider"] = "exa"
        dispatch["inputs"]["search_provider"] = "exa"
    elif change == "foreign_phase":
        raw["phase"] = "foreign"
    elif change == "foreign_case":
        raw["case_id"] = summary["case_id"] = "foreign"
    elif change == "foreign_prefix":
        raw["log_prefix"] = summary["log_prefix"] = "foreign"
    elif change == "foreign_origin":
        raw["budget_origin"] = summary["budget_origin"] = "resume_baseline"
        raw["tool_policy"]["budget_origin"] = "resume_baseline"
    elif change == "strict_budget_integer":
        raw["tool_policy"]["strict_budget"] = 1
    elif change == "raw_policy_digest_mismatch":
        raw["tool_policy"]["blocked_hosts"].append("synthetic-extra-blocked.edu")
    else:
        key, budget_key, value = {
            "max_calls_61": ("max_fetcher_calls", "max_calls", 61),
            "max_ms_120001": ("max_elapsed_ms", "max_ms", 120001),
            "max_queries_7": ("max_search_queries", "max_search_queries", 7),
            "max_queries_float": ("max_search_queries", "max_search_queries", 6.0),
        }[change]
        raw["tool_policy"][key] = raw["budgets"][budget_key] = value
    if change not in {"raw_deadline_95000", "raw_policy_digest_mismatch"}:
        summary["tool_policy"] = copy.deepcopy(raw["tool_policy"])
    dispatch["inputs"]["plan"] = json.dumps(plan)
    write(dispatch_path, dispatch)
    write(raw_path, raw)
    outer_sha = report["file_hash"](raw_path)
    summary["actions"][0]["raw_log_sha256"] = outer_sha
    summary["raw_logs"][relative] = outer_sha
    write(summary_path, summary)
    trace = pilot.read("trace.json")
    trace["actions"][0]["raw_log_sha256"] = outer_sha
    write(pilot.folder / "trace.json", trace)
    result = pilot.read("result.json")
    result["trace_sha256"] = report["digest"](trace)
    persist_result(pilot, result)
    # Keep source proof/event hashes consistent to reach the independent raw
    # policy guard; fact content, response bytes and immutable journal stay exact.
    proof = pilot.read("compact-citation-proof.json")
    for item in proof:
        item["raw_log_sha256"] = outer_sha
    write(pilot.folder / "compact-citation-proof.json", proof)
    events = pilot.read("journal-events.json")
    for event in events:
        if event["source_action_count"]:
            event["citation_proof_sha256"] = report["digest"](proof)
    write(pilot.folder / "journal-events.json", events)
    assert raw["response"] == original_response
    assert (
        summary["actions"][0]["raw_log_sha256"]
        == summary["raw_logs"][relative]
        == trace["actions"][0]["raw_log_sha256"]
        == outer_sha
    )
    with pytest.raises(ValueError, match="Original raw source identity|Original raw source policy"):
        report["verified_controller_stop"](directory)
    assert_prelabel_rejection(pilot, report, monkeypatch)
