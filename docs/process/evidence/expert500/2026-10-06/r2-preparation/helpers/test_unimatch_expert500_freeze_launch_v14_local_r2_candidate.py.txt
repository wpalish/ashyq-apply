def load(name: str, path: Path) -> Any:
    spec = importlib.util.spec_from_file_location(name, path)
    assert spec is not None and spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module

@pytest.mark.parametrize(
    "mutation",
    [
        "field_missing",
        "source_hash",
        "runtime_hash",
        "unmanifested_source",
        "unmanifested_dependency",
        "startup_pth",
        "interpreter_missing",
        "docclock",
        "doc2_off",
        "import_alias",
        "dotenv",
    ],
)
def test_runtime_changes_fail_before_any_execution(
    configuration: dict[str, Any], mutation: str
) -> None:
    runtime = copy.deepcopy(configuration["local_runtime"])
    source = Path(runtime["source_cwd"])
    deps = Path(runtime["runtime_import_roots"][0])
    if mutation == "field_missing":
        runtime.pop("runtime_files")
    elif mutation == "source_hash":
        runtime["source_manifest"][next(iter(runtime["source_manifest"]))] = "0" * 64
    elif mutation == "runtime_hash":
        runtime["runtime_files"][runtime["interpreter"]] = "0" * 64
    elif mutation == "unmanifested_source":
        (source / "app" / "new.py").write_text("PUBLIC = True")
    elif mutation in {"unmanifested_dependency", "startup_pth"}:
        (deps / ("inject.pth" if mutation == "startup_pth" else "new.py")).write_text("synthetic")
    elif mutation == "interpreter_missing":
        runtime["runtime_files"].pop(runtime["interpreter"])
    elif mutation in {"docclock", "doc2_off"}:
        relative = (
            "evaluation/research/live.py"
            if mutation == "docclock"
            else "app/adapters/documents/web_documents.py"
        )
        path = source / relative
        path.write_text(
            path.read_text()
            .replace("0.0", "1.0")
            .replace("DOCUMENT_RULES_V2 = True", "DOCUMENT_RULES_V2 = False")
        )
        runtime["source_manifest"][relative] = freeze.sha(path)
    elif mutation == "import_alias":
        alias = deps.parent / "alias"
        alias.symlink_to(deps, target_is_directory=True)
        runtime["runtime_import_roots"] = [str(alias)]
    else:
        (source / ".env").write_text("SYNTHETIC_CONTENT_NEVER_READ")
    with pytest.raises(ValueError):
        freeze.validate_runtime(runtime)

def test_concurrent_prepare_only_one_allocates(
    configuration: dict[str, Any], tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    barrier = threading.Barrier(2)
    monkeypatch.setattr(
        launch, "validate_manifest", lambda *args, **kwargs: barrier.wait(timeout=5)
    )
    pointer = write(tmp_path / "manifest.json", synthetic_manifest(configuration))
    root = tmp_path.resolve() / "concurrent"
    temp = tmp_path.resolve() / "pool"
    temp.mkdir()

    def prepare() -> bool:
        try:
            launch.prepare_trial(
                Path(pointer["path"]), repo=configuration["_repo"], root=root, temp_root=temp
            )
            return True
        except FileExistsError:
            return False

    with ThreadPoolExecutor(max_workers=2) as pool:
        outcomes = list(pool.map(lambda _number: prepare(), range(2)))
    assert sorted(outcomes) == [False, True] and len(list(temp.iterdir())) == 10


def test_parent_launch_passes_private_fd_only_and_never_inherits_env(
    trial: tuple[Path, Path, Path], monkeypatch: pytest.MonkeyPatch
) -> None:
    repo, root, _manifest = trial
    reader, writer = os.pipe()
    os.close(writer)
    seen = []

    def run(argv: list[str], **kwargs: Any) -> subprocess.CompletedProcess[str]:
        seen.append((argv, kwargs))
        assert (root / "launch-baseline-stage-start.json").is_file()
        return subprocess.CompletedProcess(argv, 0)

    monkeypatch.setattr(launch.subprocess, "run", run)
    assert launch.launch_stage(repo=repo, root=root, stage="baseline", source_key_fd=reader) == 0
    argv, kwargs = seen[0]
    assert argv[1:6] == ["-B", "-E", "-S", "-s", "-c"]
    assert str(launch.PARENT_PATH) in argv[-1] and "--source-key-fd" in argv[-1]
    assert kwargs["pass_fds"] == (reader,)
    assert kwargs["env"] == launch.read(root / "trial.json")["selector_environment"]
    assert not {"EXA_API_KEY", "UNIMATCH_EXA_API_KEY", "PYTHONPATH"} & set(kwargs["env"])
    with pytest.raises(OSError):
        os.fstat(reader)
    next_reader, next_writer = os.pipe()
    try:
        with pytest.raises(ValueError, match="previously claimed"):
            launch.launch_stage(repo=repo, root=root, stage="baseline", source_key_fd=next_reader)
        assert len(seen) == 1
    finally:
        os.close(next_reader)
        os.close(next_writer)

def test_actual_neutral_request_phase_and_dispatch_bind_same_environment(
    configuration: dict[str, Any], monkeypatch: pytest.MonkeyPatch
) -> None:
    # Evaluate only the two saved environment expressions, never the proof runner.
    tree = ast.parse(freeze.NEUTRAL_SCRIPT.read_bytes())
    probe = next(
        node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name == "run_probe"
    )
    request = next(
        node.value
        for node in ast.walk(probe)
        if isinstance(node, ast.Assign)
        and any(isinstance(target, ast.Name) and target.id == "request" for target in node.targets)
    )
    assert isinstance(request, ast.Dict)
    request_values = {
        key.value: value
        for key, value in zip(request.keys, request.values, strict=True)
        if isinstance(key, ast.Constant)
        and key.value in {"selector_environment", "selector_environment_sha256"}
    }
    phase = next(
        node
        for node in ast.walk(probe)
        if isinstance(node, ast.Call)
        and isinstance(node.func, ast.Attribute)
        and isinstance(node.func.value, ast.Name)
        and node.func.value.id == "phase_result"
        and node.func.attr == "update"
        and any(item.arg == "selector_environment" for item in node.keywords)
    )
    phase_values = {item.arg: item.value for item in phase.keywords if item.arg in request_values}
    expected = configuration["selector_environment"]
    monkeypatch.setattr(freeze, "SELECTOR_ENVIRONMENT", expected)
    namespace = {"PUBLIC_MODEL_ENVIRONMENT": expected, "digest": digest}
    for expressions in (request_values, phase_values):
        record = {
            key: eval(compile(ast.Expression(value), "<neutral-env-only>", "eval"), namespace)
            for key, value in expressions.items()
        }
        freeze.validate_neutral_environment(record)
    select = next(
        node
        for node in ast.walk(probe)
        if isinstance(node, ast.Call)
        and isinstance(node.func, ast.Name)
        and node.func.id == "select"
    )
    environment = next(item.value for item in select.keywords if item.arg == "environment")
    assert (
        eval(compile(ast.Expression(environment), "<neutral-env-only>", "eval"), namespace)
        == expected
    )

@pytest.mark.parametrize(
    "mutation",
    ["none", "missing", "digest", "codex", "sandbox", "path", "bool_integer", "configured_path"],
)
def test_fresh_cli_discovery_is_exact_and_hash_bound(
    configuration: dict[str, Any], monkeypatch: pytest.MonkeyPatch, mutation: str
) -> None:
    environment = configuration["selector_environment"]
    monkeypatch.setattr(freeze, "SELECTOR_ENVIRONMENT", dict(environment))
    proof = {
        "path": environment["PATH"],
        "codex_executable": freeze.FROZEN_CLI,
        "sandbox_executable": "/usr/bin/sandbox-exec",
        "path_matches_public_model_environment": True,
    }
    record = {"cli_discovery": proof, "cli_discovery_sha256": digest(proof)}
    if mutation == "none":
        freeze.validate_cli_discovery(record)
        return
    if mutation == "missing":
        record.pop("cli_discovery")
    elif mutation == "digest":
        record["cli_discovery_sha256"] = "0" * 64
    elif mutation == "configured_path":
        freeze.SELECTOR_ENVIRONMENT["PATH"] = "/usr/bin:/bin:/usr/sbin:/sbin"
    else:
        field, replacement = {
            "codex": ("codex_executable", "/synthetic/unpinned/codex"),
            "sandbox": ("sandbox_executable", "/synthetic/unpinned/sandbox-exec"),
            "path": ("path", "/usr/bin:/bin:/usr/sbin:/sbin"),
            "bool_integer": ("path_matches_public_model_environment", 1),
        }[mutation]
        proof[field] = replacement
        record["cli_discovery_sha256"] = digest(proof)
    with pytest.raises(ValueError, match="Fresh CLI discovery"):
        freeze.validate_cli_discovery(record)
