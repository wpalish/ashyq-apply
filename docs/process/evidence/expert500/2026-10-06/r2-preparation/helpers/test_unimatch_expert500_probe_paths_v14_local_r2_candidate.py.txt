@pytest.fixture(autouse=True)
def no_actual_calls(monkeypatch: pytest.MonkeyPatch) -> Any:
    monkeypatch.setattr(subprocess, "run", forbidden)
    monkeypatch.setattr(subprocess, "Popen", forbidden)
    monkeypatch.setattr(asyncio, "create_subprocess_exec", forbidden)
    monkeypatch.setattr(asyncio, "create_subprocess_shell", forbidden)
    monkeypatch.setattr(os, "killpg", forbidden)
    original_read = Path.read_bytes
    original_text = Path.read_text

    def allowed(path: Path) -> None:
        if (
            any(
                token in str(path).lower()
                for token in (
                    "ground_truth",
                    "identity_bindings",
                    "reviewed8",
                    "reviewed9",
                    "heldout",
                )
            )
            or path.name == ".env"
        ):
            raise AssertionError("Private inputs are forbidden")

    def read_bytes(path: Path) -> bytes:
        allowed(path)
        return original_read(path)

    def read_text(path: Path, *args: Any, **kwargs: Any) -> str:
        allowed(path)
        return original_text(path, *args, **kwargs)

    monkeypatch.setattr(Path, "read_bytes", read_bytes)
    monkeypatch.setattr(Path, "read_text", read_text)
    for path, expected in ORIGINALS.items():
        assert hashlib.sha256(Path(path).read_bytes()).hexdigest() == expected
    yield
    for path, expected in ORIGINALS.items():
        assert hashlib.sha256(Path(path).read_bytes()).hexdigest() == expected


def load(path: Path, monkeypatch: pytest.MonkeyPatch) -> Any:
    name = "_r2_probe_test_" + path.stem.replace("-", "_")
    spec = importlib.util.spec_from_file_location(name, path)
    assert spec and spec.loader
    module = importlib.util.module_from_spec(spec)
    monkeypatch.setitem(sys.modules, name, module)
    spec.loader.exec_module(module)
    return module


@dataclass
class Clock:
    elapsed_ms: int = 0

    def now(self) -> datetime:
        return datetime(2026, 10, 6, 12, tzinfo=UTC) + timedelta(milliseconds=self.elapsed_ms)

    def monotonic(self) -> float:
        return self.elapsed_ms / 1000


def events(answer: dict[str, Any]) -> str:
    return "\n".join(
        json.dumps(event)
        for event in (
            {"type": "thread.started", "thread_id": "invented-neutral-only"},
            {"type": "turn.started"},
            {
                "type": "item.completed",
                "item": {"type": "agent_message", "text": json.dumps(answer)},
            },
            {"type": "turn.completed", "usage": {"input_tokens": 1, "output_tokens": 1}},
        )
    )

def mock_processes(probe: Any, monkeypatch: pytest.MonkeyPatch) -> None:
    def run(argv: list[str], **kwargs: Any) -> Any:
        if argv == [CLI, "--version"]:
            assert set(kwargs) == {"capture_output", "text", "check", "timeout"}
            return SimpleNamespace(stdout="codex-cli 0.159.2\n")
        proof = command_proof(argv)
        assert proof["model_reasoning_effort"] is None
        assert proof["reasoning_effort_source"] == "cli_default"
        assert kwargs["env"] == probe.module.PUBLIC_MODEL_ENVIRONMENT
        assert kwargs["env"]["PATH"] == MODEL_PATH
        assert set(kwargs["env"]) == {"HOME", "PATH", "LANG", "LC_ALL", "TMPDIR"}
        assert kwargs["cwd"] == Path(argv[argv.index("-C") + 1])
        assert kwargs["timeout"] == (900000 - probe.clock.elapsed_ms) / 1000
        prompt = json.loads(kwargs["input"])
        output = Path(argv[argv.index("--output-last-message") + 1])
        phase = probe.module.PHASES[len(probe.calls)]
        journal_hash = probe.module.AcceptedJournal.initial().sha256
        if phase == "bootstrap":
            answer = {
                "kind": "action",
                "observation": "neutral probe",
                "journal_sha256": journal_hash,
                "action": {
                    "tool": "fetcher",
                    "url": "https://example.org/public-probe",
                    "query": None,
                    "domains": [],
                    "max_results": None,
                    "source_step": None,
                    "page": None,
                    "offset": None,
                    "section_offset": None,
                    "link_offset": None,
                },
                "facts": [],
                "question_findings": [],
            }
        elif phase == "data":
            answer = copy.deepcopy(prompt["synthetic_record"])
        else:
            answer = {
                "kind": "submit",
                "observation": CLOSING_OBSERVATION,
                "journal_sha256": journal_hash,
                "action": None,
                "facts": [],
                "question_findings": [],
            }
        stdout = events(answer)
        output_value = copy.deepcopy(answer)
        if probe.fault == "output_mismatch":
            output_value["observation"] = "A different atomic output."
        elif probe.fault == "wrong_hash":
            output_value["journal_sha256"] = "f" * 64
            stdout = events(output_value)
        elif probe.fault == "tool_event":
            stdout = stdout.replace('"agent_message"', '"command_execution"')
        elif probe.fault == "failed_event":
            stdout += '\n{"type":"turn.failed"}'
        elif probe.fault == "wrong_closing" and phase == "minimal_commit":
            output_value["observation"] = "A replacement closing instruction."
            stdout = events(output_value)
        output.write_text(json.dumps(output_value))
        probe.calls.append({"phase": phase, "argv": argv, "environment": kwargs["env"]})
        probe.clock.elapsed_ms += 900000 if probe.fault == "deadline" else 100
        return SimpleNamespace(stdout=stdout, stderr="", returncode=0)

    monkeypatch.setattr(subprocess, "run", run)

    assert result["turn_policy"]["wall_guard_ms"] == 900000
    assert result["turn_policy"]["max_calls"] == 60
    assert result["turn_policy"]["max_search_queries"] == 6
    assert result["requested_reasoning_effort"] is None
    assert result["reasoning_effort_source"] == "cli_default"
    assert all(item["completed_tool_free"] for item in result["results"])
    assert result["results"][0]["answer"]["facts"] == []
    assert len(result["results"][1]["answer"]["facts"]) == 4
    assert result["results"][2]["answer"]["kind"] == "submit"
    assert result["results"][2]["answer"]["facts"] == []
    assert result["results"][2]["answer"]["question_findings"] == []
