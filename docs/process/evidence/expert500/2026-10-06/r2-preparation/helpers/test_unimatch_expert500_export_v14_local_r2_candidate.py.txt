def load_module(name: str, path: Path) -> Any:
    spec = importlib.util.spec_from_file_location(name, path)
    assert spec and spec.loader
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module

def write(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, sort_keys=True, allow_nan=False))


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


@pytest.fixture(autouse=True)
def no_actual_process_or_private_input(monkeypatch: pytest.MonkeyPatch) -> None:
    def denied(*_args: Any, **_kwargs: Any) -> Any:
        raise AssertionError("Only synthetic source/process callbacks are authorized")

    monkeypatch.setattr(subprocess, "Popen", denied)
    monkeypatch.setattr(subprocess, "run", denied)
    monkeypatch.setattr(asyncio, "create_subprocess_exec", denied)
    monkeypatch.setattr(os, "killpg", denied)
    original_bytes, original_text = Path.read_bytes, Path.read_text

    def private(path: Path) -> bool:
        return any(
            token in str(path).lower()
            for token in ("ground_truth", "identity_bindings", "heldout", "/.env")
        )

    def read_bytes(path: Path) -> bytes:
        if private(path):
            raise AssertionError("Private facts or credentials must not be read")
        return original_bytes(path)

    def read_text(path: Path, *args: Any, **kwargs: Any) -> str:
        if private(path):
            raise AssertionError("Private facts or credentials must not be read")
        return original_text(path, *args, **kwargs)

    monkeypatch.setattr(Path, "read_bytes", read_bytes)
    monkeypatch.setattr(Path, "read_text", read_text)

def neutral_archive(tmp_path):
    archive = tmp_path.resolve() / "fresh-neutral"
    stream = archive / "artifacts/selector-01.events.jsonl"
    stream.parent.mkdir(parents=True)
    stream.write_bytes(b'{"type":"turn.completed"}\n')
    value = {
        "records": [
            {
                "path": stream.relative_to(archive).as_posix(),
                "original_path": "/synthetic/neutral/events.jsonl",
                "sha256": sha(stream),
                "bytes": stream.stat().st_size,
            }
        ]
    }
    write(archive / "manifest.json", value)
    pointer = {"path": str(archive / "manifest.json"), "sha256": sha(archive / "manifest.json")}
    return archive, value, pointer

@pytest.mark.parametrize(
    "bad",
    [
        "missing_stream",
        "changed_stream",
        "duplicate",
        "traversal",
        "alias",
        "wrong_size",
        "private_producer",
    ],
)
def test_neutral_archive_cannot_lose_or_substitute_original_raw_bytes(exporter, tmp_path, bad):
    archive, value, pointer = neutral_archive(tmp_path)
    stream = archive / value["records"][0]["path"]
    trial = {"neutral_originals": pointer}
    if bad == "missing_stream":
        stream.unlink()
    elif bad == "changed_stream":
        stream.write_bytes(b"changed")
    elif bad == "duplicate":
        value["records"].append(copy.deepcopy(value["records"][0]))
    elif bad == "traversal":
        value["records"][0]["path"] = "../outside"
    elif bad == "alias":
        outside = tmp_path / "outside-stream"
        outside.write_bytes(stream.read_bytes())
        stream.unlink()
        stream.symlink_to(outside)
    elif bad == "wrong_size":
        value["records"][0]["bytes"] = True
    elif bad == "private_producer":
        trial["baseline_identity_registry"] = {"path": value["records"][0]["original_path"]}
    write(archive / "manifest.json", value)
    pointer["sha256"] = sha(archive / "manifest.json")
    with pytest.raises((ValueError, FileNotFoundError)):
        exporter["neutral_originals"]({"neutral_originals": pointer}, trial)

    exporter, tmp_path, monkeypatch

    sentinel = destination / "original"
    sentinel.write_bytes(b"preserve existing byte")
    monkeypatch.setattr(
        sys,
        "argv",
        [
            str(EXPORT),
            "--root",
            str(root),
            "--protocol",
            str(protocol),
            "--destination",
            str(destination),
        ],
    )
    with pytest.raises(ValueError, match="already exists"):
        exporter["main"]()
    assert sentinel.read_bytes() == b"preserve existing byte"

def test_export_main_detects_changed_sealed_source_bytes_and_keeps_partial_original(
    exporter, copy_fixture
):
    fixture = copy_fixture
    fixture.source_log.write_bytes(b"changed after original seal")
    with pytest.raises(ValueError, match="sealed"):
        exporter["main"]()
    assert fixture.destination.is_dir()
    assert not (fixture.destination / "manifest.json").exists()
    original = fixture.destination / "submission-seal.json"
    assert original.read_bytes() == (fixture.root / "submission-seal.json").read_bytes()
