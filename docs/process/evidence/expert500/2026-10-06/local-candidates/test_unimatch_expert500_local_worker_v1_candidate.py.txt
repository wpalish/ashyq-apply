def load_module(name: str, path: Path) -> Any:
    spec = importlib.util.spec_from_file_location(name, path)
    assert spec and spec.loader
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module

class FakeClock:
    def __init__(self) -> None:
        self.value = datetime(2026, 10, 6, 12, tzinfo=UTC)
        self.start = self.value

    def now(self) -> datetime:
        return self.value

    def monotonic(self) -> float:
        return (self.value - self.start).total_seconds()

    def advance(self, seconds: float) -> None:
        self.value += timedelta(seconds=seconds)

    def guard(self) -> dict[str, Any]:
        return {
            "wall_started_at": self.start.isoformat(),
            "wall_deadline_at": (self.start + timedelta(seconds=900)).isoformat(),
            "wall_guard_ms": 900000,
        }

class FakeFetcher:
    def __init__(self, clock: FakeClock) -> None:
        self.clock = clock
        self.calls: list[tuple[str, bool]] = []
        self.robots: dict[str, str] = {}
        self.pacing: dict[str, datetime] = {}

    async def get(self, url: str, *, use_cache: bool) -> Any:
        self.calls.append((url, use_cache))
        self.robots.setdefault("example.edu", "same-object-policy")
        self.pacing["example.edu"] = self.clock.now()
        self.clock.advance(0.01)
        return SimpleNamespace(
            outcome=SimpleNamespace(value="ok"),
            content=b"<h1>Undergraduate tuition</h1><p>2027/28 fee EUR 100</p>",
            ok=True,
            status_code=200,
            final_url=url,
            url=url,
            content_type="text/html",
            from_cache=False,
            fetch_tier="http",
            error="",
            fetched_at=self.clock.now(),
        )


class FakeProvider:
    def __init__(self, clock: FakeClock) -> None:
        self.clock = clock
        self.calls = 0
        self.status: int | None = None

    async def search(self, *_args: Any, **_kwargs: Any) -> Any:
        self.calls += 1
        self.clock.advance(0.01)
        if self.status is not None:
            from app.adapters.search.base import SearchUnavailable

            raise SearchUnavailable("synthetic fixed failure", http_status=self.status)
        return SimpleNamespace(provider="exa", results=[])


def make_engine(config: Any, clock: FakeClock) -> tuple[Any, FakeFetcher, FakeProvider]:
    fetcher, provider = FakeFetcher(clock), FakeProvider(clock)
    gate = admission_module.SourceAdmission(
        config.admission_root, run_id=config.admission_run_id, provider="exa", now=clock.now
    )

    async def cold_get(url: str) -> Any:
        return await fetcher.get(url, use_cache=False)

    engine = worker.PersistentWorker(
        config,
        str(uuid.uuid4()),
        execute=controller.execute,
        get=cold_get,
        search=provider.search,
        admission=gate,
        clock=clock.now,
        monotonic=clock.monotonic,
    )
    startup(config, engine.session_id)
    worker.atomic_new(config.session_root / "worker.stdout", b"")
    return engine, fetcher, provider


def deliver(engine: Any, payload: dict[str, Any], *, prior: Any = None, turn: int = 1) -> Any:
    destination = engine.config.delivery_root / f"controller-{turn:02d}"
    destination.mkdir(mode=0o700)
    request = {
        "schema_version": "1",
        "transport_kind": worker.TRANSPORT,
        "session_id": engine.session_id,
        "local_run_id": str(uuid.uuid4()),
        "sequence": engine.sequence + 1,
        "case_id": engine.config.case_id,
        "stage": engine.config.stage,
        "source_sha": engine.config.source_sha,
        "session_config_sha256": worker.digest(engine.config.as_dict()),
        "wall_guard": engine.config.wall_guard,
        "prior_delivery_sha256": prior.receipt_sha256 if prior else None,
        "payload": payload,
    }
    worker.atomic_new(destination / "request.json", worker.canonical_bytes(request))
    frame = asyncio.run(engine.deliver(request, destination))
    data = worker.canonical_bytes(frame) + b"\n"
    stream = engine.config.session_root / "worker.stdout"
    offset = stream.stat().st_size
    with stream.open("ab") as output:
        output.write(data)
    delivery = worker._publish_local_receipt(
        destination,
        data,
        engine.config,
        stdout_range={
            "offset": offset,
            "length": len(data),
            "sha256": worker.hashlib.sha256(data).hexdigest(),
        },
    )
    engine.acknowledge(delivery.receipt_sha256)
    return delivery

def test_persistent_stock_controller_exact_resume_cold_reads_and_state(config: Any) -> None:
    clock = FakeClock()
    engine, fetcher, provider = make_engine(config, clock)
    first = deliver(engine, plan(config))
    assert validate(first, config).source_counters == {
        "fetcher_calls": 1,
        "search_queries": 0,
        "elapsed_ms": 10,
    }
    clock.advance(180)  # model/transport waiting never enters source_ms
    second = deliver(engine, plan(config, first), prior=first, turn=100)
    verified = validate(second, config, first)
    assert verified.source_counters["fetcher_calls"] == 2
    assert verified.source_counters["search_queries"] == 0
    assert 20 <= verified.source_counters["elapsed_ms"] <= 21
    assert verified.source_counters == {
        key: second.summary["final"][key]
        for key in ("fetcher_calls", "search_queries", "elapsed_ms")
    }
    assert fetcher.calls == [("https://example.edu/course", False)] * 2
    assert fetcher.robots == {"example.edu": "same-object-policy"}
    assert provider.calls == 0
    assert second.summary["raw_logs"].keys() > first.summary["raw_logs"].keys()
    assert (
        worker.selector_environment(worker.source_environment(config, "synthetic-key"))
        == config.public_environment
    )

                    "provider": "exa",

class FakeProcess:
    def __init__(self, *, config: Any, clock: FakeClock, hang: bool = False) -> None:
        self.config, self.clock, self.hang = config, clock, hang
        self.pid = 4567
        self.stdout = FakeStdout()
        self.stdin = FakeStdin(self.command)
        self.returncode: int | None = None
        self.wait_calls = 0
        bootstrap = worker.load_json((config.session_root / "session-bootstrap.json").read_bytes())
        session_id = bootstrap["session_id"]
        startup(config, session_id, pid=self.pid)
        fetcher = FakeFetcher(clock)
        provider = FakeProvider(clock)
        gate = admission_module.SourceAdmission(
            config.admission_root, run_id=config.admission_run_id, provider="exa", now=clock.now
        )

        async def cold(url: str) -> Any:
            return await fetcher.get(url, use_cache=False)

        self.engine = worker.PersistentWorker(
            config,
            session_id,
            execute=controller.execute,
            get=cold,
            search=provider.search,
            admission=gate,
            clock=clock.now,
            monotonic=clock.monotonic,
        )
        self.stdout.lines.put(
            worker.canonical_bytes({"kind": "ready", "session_id": session_id}) + b"\n"
        )

    def command(self, value: dict[str, Any]) -> None:
        if value["kind"] == "close":
            self.returncode = 0
            self.stdout.lines.put(b"")
        elif value["kind"] == "acknowledge":
            self.engine.acknowledge(value["receipt_sha256"])
        elif self.hang:
            self.clock.advance(901)
        else:
            destination = Path(value["destination"])
            frame = asyncio.run(
                self.engine.deliver(
                    worker.load_json((destination / "request.json").read_bytes()), destination
                )
            )
            self.stdout.lines.put(worker.canonical_bytes(frame) + b"\n")

    def poll(self) -> int | None:
        return self.returncode

    def wait(self, timeout: float) -> int:
        self.wait_calls += 1
        if self.returncode is None:
            raise subprocess.TimeoutExpired("synthetic source process", timeout)
        return self.returncode

@pytest.mark.parametrize("mutation", ["missing", "extra", "hash", "source", "imports", "symlink"])
def test_baseline_private_input_is_exact_opaque_and_outside_public_roots(
    config: Any, mutation: str
) -> None:
    baseline = baseline_config(config)
    pointer = dict(baseline.baseline_identity_registry)
    if mutation == "missing":
        replacement = None
    else:
        if mutation == "extra":
            pointer["contents"] = "synthetic private contents must not be serialized"
        elif mutation == "hash":
            pointer["sha256"] = "0" * 64
        elif mutation in {"source", "imports", "symlink"}:
            private_path = Path(pointer["path"])
            if mutation == "source":
                moved = baseline.source_cwd / "private-input.json"
                moved.write_bytes(private_path.read_bytes())
            elif mutation == "imports":
                moved = baseline.runtime_import_roots[0] / "private-input.json"
                moved.write_bytes(private_path.read_bytes())
            else:
                moved = private_path.parent / "aliased-private-input.json"
                moved.symlink_to(private_path)
            pointer["path"] = str(moved)
        replacement = pointer
    with pytest.raises(ValueError):
        replace(baseline, baseline_identity_registry=replacement).validate()

def unavailable_delivery(
    config: Any, monkeypatch: pytest.MonkeyPatch, *, with_prior: bool = False, early: bool = False
) -> tuple[Any, Any]:
    clock = FakeClock()

    class InterruptedProcess(FakeProcess):
        def command(self, value: dict[str, Any]) -> None:
            if value["kind"] == "deliver" and self.hang:
                artifact = Path(value["destination"]) / "artifact"
                artifact.mkdir()
                (artifact / "original-partial.txt").write_bytes(b"synthetic interrupted original")
                if not early:
                    clock.advance(901)
                return
            super().command(value)

    process: InterruptedProcess | None = None

    def factory(_argv: Any, **_kwargs: Any) -> InterruptedProcess:
        nonlocal process
        process = InterruptedProcess(config=config, clock=clock)
        return process

    def kill(_pid: int, sig: int) -> None:
        assert process is not None
        if sig == signal.SIGKILL:
            process.returncode = -sig
            process.stdout.lines.put(b"")

    session = worker.start_session(
        config,
        private_key="synthetic-key",
        process_factory=factory,
        clock=clock.now,
        monotonic=clock.monotonic,
        kill_group=kill,
    )
    prior = (
        session.execute(
            plan(config), config.delivery_root / "controller-01", wall_guard=config.wall_guard
        )
        if with_prior
        else None
    )
    assert process is not None
    process.hang = True
    monkeypatch.setattr(
        session,
        "_receive",
        lambda _timeout: (_ for _ in ()).throw(TimeoutError("synthetic source wait")),
    )
    delivery = session.execute(
        plan(config, prior),
        config.delivery_root / ("controller-02" if prior else "controller-01"),
        wall_guard=config.wall_guard,
        prior_delivery=prior,
    )
    return delivery, prior


@pytest.mark.parametrize("with_prior", [False, True])
def test_unknown_guard_tail_preserves_only_original_prefix_and_partial_bytes(
    config: Any, monkeypatch: pytest.MonkeyPatch, with_prior: bool
) -> None:
    delivery, prior = unavailable_delivery(config, monkeypatch, with_prior=with_prior)
    verified = worker.validate_unavailable_delivery(
        delivery.receipt_path.parent,
        expected_case_id=config.case_id,
        expected_stage=config.stage,
        expected_source_sha=config.source_sha,
        expected_wall_guard=config.wall_guard,
        prior_delivery=prior,
    )
    assert verified == delivery
    assert verified.receipt["terminal_reason"] == "wall_guard_exhausted"
    assert (
        verified.source_counters is None and verified.summary is None and verified.capture is None
    )
    assert set(verified.receipt["partial_artifacts"]) == {"original-partial.txt"}
    if prior:
        assert prior.source_counters["fetcher_calls"] == 1
        assert verified.receipt["prior_delivery_sha256"] == prior.receipt_sha256
    with pytest.raises(ValueError, match="Unknown"):
        validate(verified, config, prior)


@pytest.mark.parametrize(
    "mutation",
    [
        "case",
        "guard",
        "count",
        "flag",
        "request",
        "startup",
        "stream",
        "partial",
        "reaped",
        "returncode",
        "prior",
        "pid",
    ],
)
def test_unknown_source_receipt_binding_cannot_be_reclassified_by_editable_metadata(
    config: Any, monkeypatch: pytest.MonkeyPatch, mutation: str
) -> None:
    delivery, prior = unavailable_delivery(config, monkeypatch, with_prior=True)
    receipt = worker.load_json(delivery.receipt_path.read_bytes())
    if mutation == "case":
        receipt["case_id"] = "other"
    elif mutation == "guard":
        receipt["wall_guard"]["wall_deadline_at"] = "2026-10-06T13:00:00+00:00"
    elif mutation == "count":
        receipt["source_counters"] = {"fetcher_calls": 1, "search_queries": 0, "elapsed_ms": 10}
    elif mutation == "flag":
        receipt["source_counters_known"] = 0
    elif mutation == "request":
        receipt["request_sha256"] = "0" * 64
    elif mutation == "startup":
        receipt["session_start_sha256"] = "0" * 64
    elif mutation == "stream":
        (config.session_root / "worker.stdout").write_bytes(b"forged terminal stream")
    elif mutation == "partial":
        (delivery.receipt_path.parent / "artifact/original-partial.txt").write_bytes(b"changed")
    elif mutation == "reaped":
        receipt["terminal_process"]["reaped"] = False
    elif mutation == "returncode":
        receipt["terminal_process"]["returncode"] = None
    elif mutation == "prior":
        receipt["prior_delivery_sha256"] = None
    elif mutation == "pid":
        receipt["terminal_process"]["pid"] += 1
    delivery.receipt_path.write_bytes(worker.canonical_bytes(receipt))
    with pytest.raises(ValueError):
        worker.validate_unavailable_delivery(
            delivery.receipt_path.parent,
            expected_case_id=config.case_id,
            expected_stage=config.stage,
            expected_source_sha=config.source_sha,
            expected_wall_guard=config.wall_guard,
            prior_delivery=prior,
        )
